131static void shuffle_indices(size_t* indices, size_t size) {
132 for (size_t i = size - 1; i > 0; i--) {
133 size_t j = rand() % (i + 1);
134 size_t temp = indices[i];

140double evaluate_model(Model* model,
141 const double* data, const double* labels,
142 size_t num_samples, size_t input_size, size_t output_size,
143 LossType loss_type) {
144 if (!model || !data || !labels) return 0.0;
145
146 double total_loss = 0.0;
147 size_t batch_size = 32;
148
149 double* predictions = malloc(batch_size * output_size * sizeof(double));
150 if (!predictions) return 0.0;
151
152 for (size_t i = 0; i < num_samples; i += batch_size) {
153 size_t current_batch = (i + batch_size > num_samples) ?
154 (num_samples - i) : batch_size;
155
156 const double* batch_data = data + i * input_size;
157 const double* batch_labels = labels + i * output_size;
158
159 // Forward pass
160 model->forward(model->model_data, batch_data, predictions,
161 current_batch, input_size, output_size);
162
163 // Compute loss
164 double batch_loss = 0.0;
165 switch (loss_type) {
166 case LOSS_MSE:
167 batch_loss = loss_mse_forward(predictions, batch_labels,
168 current_batch * output_size);
169 break;
170 case LOSS_CROSS_ENTROPY:
171 batch_loss = loss_cross_entropy_forward(predictions, batch_labels,
172 current_batch, output_size);
173 break;
174 case LOSS_BINARY_CROSS_ENTROPY:
175 batch_loss = loss_binary_cross_entropy_forward(predictions, batch_labels,
176 current_batch * output_size);
177 break;
178 case LOSS_MAE:
179 batch_loss = loss_mae_forward(predictions, batch_labels,
180 current_batch * output_size);
181 break;
182 case LOSS_HUBER:
183 batch_loss = loss_huber_forward(predictions, batch_labels,
184 current_batch * output_size, 1.0);
185 break;
186 }
187
188 total_loss += batch_loss * current_batch;
189 }
190
191 free(predictions);
192 return total_loss / num_samples;
193}

195TrainingHistory* train_model(Model* model,
196 const double* train_data, const double* train_labels,
197 size_t num_train_samples, size_t input_size, size_t output_size,
198 TrainingConfig config) {
199 if (!model || !train_data || !train_labels) return NULL;
200
201 // Create training history
202 TrainingHistory* history = training_history_create(config.num_epochs);
203 if (!history) return NULL;
204
205 // Get model parameters
206 size_t num_params;
207 double* params = model->get_params(model->model_data, &num_params);
208
209 // Create optimizer
210 void* optimizer = NULL;
211 SGDOptimizer* sgd_opt = NULL;
212 AdamOptimizer* adam_opt = NULL;
213 RMSpropOptimizer* rmsprop_opt = NULL;
214
215 switch (config.optimizer_type) {
216 case OPT_SGD: {
217 SGDConfig sgd_config = sgd_default_config();
218 sgd_config.learning_rate = config.learning_rate;
219 sgd_opt = sgd_create(num_params, sgd_config);
220 optimizer = sgd_opt;
221 break;
222 }
223 case OPT_ADAM: {
224 AdamConfig adam_config = adam_default_config();
225 adam_config.learning_rate = config.learning_rate;
226 adam_opt = adam_create(num_params, adam_config);
227 optimizer = adam_opt;
228 break;
229 }
230 case OPT_RMSPROP: {
231 RMSpropConfig rmsprop_config = rmsprop_default_config();
232 rmsprop_config.learning_rate = config.learning_rate;
233 rmsprop_opt = rmsprop_create(num_params, rmsprop_config);
234 optimizer = rmsprop_opt;
235 break;
236 }
237 }
238
239 if (!optimizer) {
240 training_history_destroy(history);
241 return NULL;
242 }
243
244 // Create learning rate scheduler if needed
245 LRScheduler* scheduler = NULL;
246 if (config.use_scheduler) {
247 scheduler = lr_scheduler_create(config.scheduler_config);
248 }
249
250 // Allocate batch buffers
251 double* batch_data = malloc(config.batch_size * input_size * sizeof(double));
252 double* batch_labels = malloc(config.batch_size * output_size * sizeof(double));
253 double* predictions = malloc(config.batch_size * output_size * sizeof(double));
254 double* grad_output = malloc(config.batch_size * output_size * sizeof(double));
255
256 if (!batch_data || !batch_labels || !predictions || !grad_output) {
257 free(batch_data);
258 free(batch_labels);
259 free(predictions);
260 free(grad_output);
261 training_history_destroy(history);
262 if (sgd_opt) sgd_destroy(sgd_opt);
263 if (adam_opt) adam_destroy(adam_opt);
264 if (rmsprop_opt) rmsprop_destroy(rmsprop_opt);
265 if (scheduler) lr_scheduler_destroy(scheduler);
266 return NULL;
267 }
268
269 // Create indices for shuffling
270 size_t* indices = malloc(num_train_samples * sizeof(size_t));
271 for (size_t i = 0; i < num_train_samples; i++) {
272 indices[i] = i;
273 }
274
275 // Training loop
276 srand(time(NULL));
277
278 for (size_t epoch = 0; epoch < config.num_epochs; epoch++) {
279 double epoch_loss = 0.0;
280 size_t num_batches = 0;
281
282 // Shuffle data if needed
283 if (config.shuffle) {
284 shuffle_indices(indices, num_train_samples);
285 }
286
287 // Batch training
288 for (size_t i = 0; i < num_train_samples; i += config.batch_size) {
289 size_t current_batch = (i + config.batch_size > num_train_samples) ?
290 (num_train_samples - i) : config.batch_size;
291
292 // Gather batch data
293 for (size_t b = 0; b < current_batch; b++) {
294 size_t idx = indices[i + b];
295 memcpy(batch_data + b * input_size,
296 train_data + idx * input_size,
297 input_size * sizeof(double));
298 memcpy(batch_labels + b * output_size,
299 train_labels + idx * output_size,
300 output_size * sizeof(double));
301 }
302
303 // Zero gradients
304 model->zero_grad(model->model_data);
305
306 // Forward pass
307 model->forward(model->model_data, batch_data, predictions,
308 current_batch, input_size, output_size);
309
310 // Compute loss and gradients
311 LossResult* loss_result = NULL;
312 switch (config.loss_type) {
313 case LOSS_MSE:
314 loss_result = loss_mse(predictions, batch_labels,
315 current_batch * output_size);
316 break;
317 case LOSS_CROSS_ENTROPY:
318 loss_result = loss_cross_entropy(predictions, batch_labels,
319 current_batch, output_size);
320 break;
321 case LOSS_BINARY_CROSS_ENTROPY:
322 loss_result = loss_binary_cross_entropy(predictions, batch_labels,
323 current_batch * output_size);
324 break;
325 case LOSS_MAE:
326 loss_result = loss_mae(predictions, batch_labels,
327 current_batch * output_size);
328 break;
329 case LOSS_HUBER:
330 loss_result = loss_huber(predictions, batch_labels,
331 current_batch * output_size, 1.0);
332 break;
333 }
334
335 if (loss_result) {
336 epoch_loss += loss_result->loss * current_batch;
337
338 // Backward pass
339 memcpy(grad_output, loss_result->gradients,
340 current_batch * output_size * sizeof(double));
341 model->backward(model->model_data, grad_output, NULL,
342 current_batch, input_size, output_size);
343
344 loss_result_destroy(loss_result);
345 }
346
347 // Optimizer step
348 size_t num_grads;
349 double* grads = model->get_grads(model->model_data, &num_grads);
350
351 switch (config.optimizer_type) {
352 case OPT_SGD:
353 sgd_step(sgd_opt, params, grads);
354 break;
355 case OPT_ADAM:
356 adam_step(adam_opt, params, grads);
357 break;
358 case OPT_RMSPROP:
359 rmsprop_step(rmsprop_opt, params, grads);
360 break;
361 }
362
363 num_batches++;
364 }
365
366 epoch_loss /= num_train_samples;
367
368 // Update learning rate scheduler
369 double current_lr = config.learning_rate;
370 if (scheduler) {
371 lr_scheduler_step(scheduler);
372 current_lr = lr_scheduler_get_lr(scheduler);
373
374 // Update optimizer learning rate
375 switch (config.optimizer_type) {
376 case OPT_SGD:
377 sgd_opt->config.learning_rate = current_lr;
378 break;
379 case OPT_ADAM:
380 adam_opt->config.learning_rate = current_lr;
381 break;
382 case OPT_RMSPROP:
383 rmsprop_opt->config.learning_rate = current_lr;
384 break;
385 }
386 }
387
388 // Update history
389 training_history_update(history, epoch, epoch_loss, 0.0, 0.0, 0.0, current_lr);
390
391 // Print progress
392 if (config.verbose && (epoch + 1) % config.print_every == 0) {
393 printf("Epoch %zu/%zu - Loss: %.6f - LR: %.6f\n",
394 epoch + 1, config.num_epochs, epoch_loss, current_lr);
395 }
396 }
397
398 // Cleanup
399 free(indices);
400 free(batch_data);
401 free(batch_labels);
402 free(predictions);
403 free(grad_output);
404
405 if (sgd_opt) sgd_destroy(sgd_opt);
406 if (adam_opt) adam_destroy(adam_opt);
407 if (rmsprop_opt) rmsprop_destroy(rmsprop_opt);
408 if (scheduler) lr_scheduler_destroy(scheduler);
409
410 return history;
411}
