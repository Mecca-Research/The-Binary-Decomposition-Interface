48// --- Constraint Solving ---

50bool type_inference_solve(TypeInferenceContext* ctx) {
51 // Simplified constraint solver
52 bool changed = true;
53 int iterations = 0;
54 const int max_iterations = 100;
55
56 while (changed && iterations < max_iterations) {
57 changed = false;
58 iterations++;
59
60 for (size_t i = 0; i < ctx->constraints.len; i++) {
61 TypeConstraint* c = &ctx->constraints.data[i];
62
63 switch (c->kind) {
64 case CONSTRAINT_EQUALITY:
65 if (unify_types(ctx, c->left, c->right)) {
66 changed = true;
67 }
68 break;
69
70 case CONSTRAINT_SUBTYPE:
71 // Simplified subtype checking
72 if (bci_type_ext_is_assignable(c->left, c->right)) {
73 changed = true;
74 }
75 break;
76
77 default:
78 break;
79 }
80 }
81 }
82
83 return iterations < max_iterations;
84}

113BciTypeExt* infer_lambda_type(TypeInferenceContext* ctx, AstNode* lambda) {
114 if (!ctx || !lambda || lambda->kind != AST_NODE_LAMBDA) return nullptr;
115
116 // Create function type
117 BciTypeExt* return_type = nullptr; // Infer from body
118 BciTypeExt* func_type = bci_type_function_create(return_type, false);
119
120 // Add parameter types
121 for (size_t i = 0; i < lambda->as.lambda.params.len; i++) {
122 AstVarDecl* param = lambda->as.lambda.params.data[i];
123 BciTypeExt* param_type = (BciTypeExt*)param->type;
124 bci_type_function_add_param(func_type, param_type);
125 }
126
127 return func_type;
128}
