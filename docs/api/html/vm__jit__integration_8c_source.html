26// JIT Code Cache Implementation
27JITCodeCache* jit_code_cache_create(size_t max_entries) {
28 JITCodeCache* cache = (JITCodeCache*)calloc(1, sizeof(JITCodeCache));
29 if (!cache) return NULL;
30
31 cache->bucket_count = max_entries / 4; // Load factor of ~4
32 if (cache->bucket_count < 16) cache->bucket_count = 16;
33
34 cache->buckets = (JITCacheEntry**)calloc(cache->bucket_count, sizeof(JITCacheEntry*));
35 if (!cache->buckets) {
36 free(cache);
37 return NULL;
38 }
39
40 cache->max_entries = max_entries;
41 cache->entry_count = 0;
42 cache->cache_hits = 0;
43 cache->cache_misses = 0;
44
45 return cache;
46}

48void jit_code_cache_destroy(JITCodeCache* cache) {

68static size_t hash_function_id(uint32_t function_id, size_t bucket_count) {
69 return function_id % bucket_count;

72CompiledCode* jit_code_cache_get(JITCodeCache* cache, uint32_t function_id) {
73 if (!cache) return NULL;
74
75 size_t bucket = hash_function_id(function_id, cache->bucket_count);
76 JITCacheEntry* entry = cache->buckets[bucket];
77
78 while (entry) {
79 if (entry->function_id == function_id) {
80 cache->cache_hits++;
81 entry->last_used = (uint64_t)time(NULL);
82 return entry->compiled_code;
83 }
84 entry = entry->next;
85 }
86
87 cache->cache_misses++;
88 return NULL;
89}

91bool jit_code_cache_put(JITCodeCache* cache, uint32_t function_id, CompiledCode* code) {
92 if (!cache || !code) return false;
93
94 // Check if already exists
95 if (jit_code_cache_get(cache, function_id)) {
96 return true; // Already cached
97 }
98
99 // Check capacity
100 if (cache->entry_count >= cache->max_entries) {
101 // TODO: Implement LRU eviction
102 return false;
103 }
104
105 // Create new entry
106 JITCacheEntry* entry = (JITCacheEntry*)malloc(sizeof(JITCacheEntry));
107 if (!entry) return false;
108
109 entry->function_id = function_id;
110 entry->compiled_code = code;
111 entry->last_used = (uint64_t)time(NULL);
112
113 // Insert into hash table
114 size_t bucket = hash_function_id(function_id, cache->bucket_count);
115 entry->next = cache->buckets[bucket];
116 cache->buckets[bucket] = entry;
117
118 cache->entry_count++;
119 return true;
120}

122void jit_code_cache_remove(JITCodeCache* cache, uint32_t function_id) {
123 if (!cache) return;
124
125 size_t bucket = hash_function_id(function_id, cache->bucket_count);
126 JITCacheEntry** entry_ptr = &cache->buckets[bucket];
127
128 while (*entry_ptr) {
129 JITCacheEntry* entry = *entry_ptr;
130 if (entry->function_id == function_id) {
131 *entry_ptr = entry->next;
132 if (entry->compiled_code) {
133 compiled_code_destroy(entry->compiled_code);
134 }
135 free(entry);
136 cache->entry_count--;

168JITIntegratedVM* jit_vm_create_with_config(size_t heap_size, const JITConfig* config) {
169 JITIntegratedVM* vm = (JITIntegratedVM*)calloc(1, sizeof(JITIntegratedVM));
170 if (!vm) return NULL;
171
172 // Create base enhanced VM
173 vm->base_vm = enhanced_vm_create(heap_size);
174 if (!vm->base_vm) {
175 free(vm);
176 return NULL;
177 }
178
179 // Copy configuration
180 if (config) {
181 vm->config = *config;
182 } else {
183 vm->config = jit_vm_default_config();
184 }
185
186 // Initialize JIT components if enabled
187 if (vm->config.enable_jit) {
188 // Create JIT compiler
189 vm->jit_compiler = jit_compiler_create();
190 if (!vm->jit_compiler) {
191 enhanced_vm_destroy(vm->base_vm);
192 free(vm);
193 return NULL;
194 }
195
196 if (jit_compiler_init(vm->jit_compiler) != JIT_STATUS_SUCCESS) {
197 jit_compiler_destroy(vm->jit_compiler);
198 enhanced_vm_destroy(vm->base_vm);
199 free(vm);
200 return NULL;
201 }
202
203 // Create hot path detector if enabled
204 if (vm->config.enable_hotspot_detection) {
205 vm->hot_path_detector = hot_path_detector_create();
206 if (vm->hot_path_detector) {
207 hot_path_detector_set_thresholds(
208 vm->hot_path_detector,
209 vm->config.jit_threshold,
210 vm->config.optimization_threshold
211 );
212 }
213 }
214
215 // Create tiered compilation manager if enabled
216 if (vm->config.enable_tiered_compilation && vm->jit_compiler && vm->hot_path_detector) {
217 vm->tiered_compilation = tiered_compilation_create(vm->jit_compiler, vm->hot_path_detector);
218 if (vm->tiered_compilation) {
219 tiered_compilation_set_thresholds(
220 vm->tiered_compilation,
221 vm->config.jit_threshold,
222 vm->config.optimization_threshold
223 );
224 }
225 }
226
227 // Create code cache
228 vm->code_cache = jit_code_cache_create(vm->config.max_compiled_functions);
229 }
230
231 // Initialize statistics
232 vm->function_executions = 0;
233 vm->jit_compilations = 0;
234 vm->jit_optimizations = 0;
235 vm->jit_execution_time_ns = 0;
236 vm->interpreter_execution_time_ns = 0;
237
238 return vm;
239}

275JITVmResult jit_vm_execute(JITIntegratedVM* vm, const Chunk* chunk) {
276 JITVmResult result = {0};
277
278 if (!vm || !chunk) {
279 result.success = false;
280 return result;
281 }
282
283 vm->function_executions++;
284
285 // For simplicity, use function_id = 0 for chunk execution
286 uint32_t function_id = 0;
287
288 uint64_t start_time = get_time_ns();
289
290 // Check if JIT is enabled and function is compiled
291 CompiledCode* compiled_code = NULL;
292 if (vm->config.enable_jit && vm->code_cache) {
293 compiled_code = jit_code_cache_get(vm->code_cache, function_id);
294 }
295
296 if (compiled_code) {
297 // Execute JIT-compiled code
298 int64_t jit_result = 0;
299 int64_t args[] = {0}; // Mock arguments
300
301 JITStatus status = jit_compiler_execute(
302 vm->jit_compiler,
303 compiled_code,
304 vm->base_vm,
305 args,
306 0,
307 &jit_result
308 );
309
310 uint64_t end_time = get_time_ns();
311 result.execution_time_ns = end_time - start_time;
312 vm->jit_execution_time_ns += result.execution_time_ns;
313
314 result.success = (status == JIT_STATUS_SUCCESS);
315 result.result_value = (double)jit_result;
316 result.used_jit = true;
317 result.compilation_time_ns = 0;
318
319 // Record execution for hot path detection
320 if (vm->hot_path_detector) {
321 hot_path_detector_record_execution(
322 vm->hot_path_detector,
323 function_id,
324 0, // basic_block_id
325 result.execution_time_ns
326 );
327 }
328 } else {
329 // Execute with interpreter
330 EnhancedVmResult vm_result = enhanced_vm_execute_with_result(vm->base_vm, chunk);
331
332 uint64_t end_time = get_time_ns();
333 result.execution_time_ns = end_time - start_time;
334 vm->interpreter_execution_time_ns += result.execution_time_ns;
335
336 result.success = vm_result.success;
337 result.result_value = vm_result.result_value;
338 result.used_jit = false;
339 result.compilation_time_ns = 0;
340
341 // Record execution for hot path detection
342 if (vm->hot_path_detector) {
343 hot_path_detector_record_execution(
344 vm->hot_path_detector,
345 function_id,
346 0,
347 result.execution_time_ns
348 );
349
350 // Check if function should be compiled
351 if (hot_path_detector_is_hot(vm->hot_path_detector, function_id, 0) &&
352 !compiled_code && vm->jit_compiler) {
353
354 uint64_t compile_start = get_time_ns();
355 if (jit_vm_compile_function(vm, function_id, chunk)) {
356 uint64_t compile_end = get_time_ns();
357 result.compilation_time_ns = compile_end - compile_start;
358 vm->jit_compilations++;
359 }
360 }
361 }
362 }
363
364 return result;
365}

442// Statistics
443void jit_vm_get_stats(const JITIntegratedVM* vm, JITVmStats* stats) {
444 if (!vm || !stats) return;
445
446 memset(stats, 0, sizeof(JITVmStats));
447
448 stats->total_executions = vm->function_executions;
449 stats->jit_compilations = vm->jit_compilations;
450 stats->jit_optimizations = vm->jit_optimizations;
451 stats->jit_execution_time_ns = vm->jit_execution_time_ns;
452 stats->interpreter_execution_time_ns = vm->interpreter_execution_time_ns;
453
454 // Get base VM stats
455 uint64_t total_exec, jit_exec, interp_exec;
456 enhanced_vm_get_stats(vm->base_vm, &total_exec, &jit_exec, &interp_exec);
457
458 stats->jit_executions = jit_exec;
459 stats->interpreter_executions = interp_exec;
460
461 // Get cache stats
462 if (vm->code_cache) {
463 stats->cache_hits = vm->code_cache->cache_hits;
464 stats->cache_misses = vm->code_cache->cache_misses;
465 }
466
467 // Calculate speedup ratio
468 if (stats->interpreter_execution_time_ns > 0 && stats->jit_execution_time_ns > 0) {
469 stats->jit_speedup_ratio = (double)stats->interpreter_execution_time_ns /
470 (double)stats->jit_execution_time_ns;
471 } else {
472 stats->jit_speedup_ratio = 1.0;
473 }
474
475 // Get compilation time from JIT compiler
476 if (vm->jit_compiler) {
477 uint64_t functions_compiled, compilation_time, optimization_time;
478 jit_compiler_get_stats(vm->jit_compiler, &functions_compiled, &compilation_time, &optimization_time);
479 stats->compilation_time_ns = compilation_time + optimization_time;
480 }
481}
