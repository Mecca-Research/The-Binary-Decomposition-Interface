140// Patch a jump instruction
141void codegen_patch_jump(CodeGenerator* codegen, int offset) {
142 Chunk* chunk = codegen->current_function->chunk;
143 int jump = chunk->count - offset - 2;
144
145 if (jump > UINT16_MAX) {
146 codegen_error(codegen, "Too much code to jump over");
147 return;
148 }
149
150 chunk->code[offset] = (jump >> 8) & 0xff;
151 chunk->code[offset + 1] = jump & 0xff;
152}

172// End current scope
173void codegen_end_scope(CodeGenerator* codegen) {
174 FunctionContext* function = codegen->current_function;
175 function->scope_depth--;
176
177 // Pop local variables that go out of scope
178 while (function->local_count > 0 &&
179 function->locals[function->local_count - 1].depth > function->scope_depth) {
180
181 if (function->locals[function->local_count - 1].is_captured) {
182 codegen_emit_byte(codegen, OPCODE_CLOSE_UPVALUE);
183 } else {
184 codegen_emit_byte(codegen, OPCODE_POP);
185 }
186 function->local_count--;
187 }
188}

220// Add an upvalue
221static int add_upvalue(FunctionContext* function, uint8_t index, bool is_local) {
222 // Check if upvalue already exists
223 for (int i = 0; i < function->upvalue_count; i++) {
224 Upvalue* upvalue = &function->upvalues[i];
225 if (upvalue->index == index && upvalue->is_local == is_local) {
226 return i;
227 }
228 }
229
230 if (function->upvalue_count >= 256) {
231 return -1; // Too many upvalues
232 }
233
234 function->upvalues[function->upvalue_count].is_local = is_local;
235 function->upvalues[function->upvalue_count].index = index;
236 return function->upvalue_count++;
237}

239// Resolve an upvalue (captured variable)
240int codegen_resolve_upvalue(CodeGenerator* codegen, FunctionContext* function, const char* name) {
241 if (function->enclosing == NULL) {
242 return -1; // No enclosing function
243 }
244
245 // Try to resolve in enclosing function's locals
246 int local = -1;
247 FunctionContext* enclosing = function->enclosing;
248 for (int i = enclosing->local_count - 1; i >= 0; i--) {
249 if (strcmp(enclosing->locals[i].name, name) == 0) {
250 local = i;
251 break;
252 }
253 }
254
255 if (local != -1) {
256 enclosing->locals[local].is_captured = true;
257 return add_upvalue(function, (uint8_t)local, true);
258 }
259
260 // Try to resolve in enclosing function's upvalues
261 int upvalue = codegen_resolve_upvalue(codegen, enclosing, name);
262 if (upvalue != -1) {
263 return add_upvalue(function, (uint8_t)upvalue, false);
264 }
265
266 return -1; // Not found
267}

304
305 // For now, only handle numeric literals

314 literal->type->kind == BCI_TYPE_U64)) {
315 codegen_emit_constant(codegen, (double)literal->value.i64);
316 } else if (literal->type && (literal->type->kind == BCI_TYPE_F32 ||

325// Generate code for a binary operation
326void codegen_binary_op(CodeGenerator* codegen, AstNode* node) {
327 assert(node->kind == AST_NODE_BINARY_OP);
328
329 AstBinaryOp* binop = &node->as.binary_op;
330
331 // Handle short-circuit operators specially
332 if (strcmp(binop->op, "&&") == 0) {
333 // Short-circuit AND: left && right
334 // Evaluate left operand first
335 codegen_node(codegen, binop->left);
336
337 // If left is false, skip right evaluation and keep false on stack
338 int end_jump = codegen_emit_jump(codegen, OPCODE_JUMP_IF_FALSE);
339
340 // Left was true, pop it and evaluate right
341 codegen_emit_byte(codegen, OPCODE_POP);
342 codegen_node(codegen, binop->right);
343
344 // Patch the jump to here (if left was false, we skip right and keep left's false value)
345 codegen_patch_jump(codegen, end_jump);
346 return;
347 } else if (strcmp(binop->op, "||") == 0) {
348 // Short-circuit OR: left || right
349 // Evaluate left operand first
350 codegen_node(codegen, binop->left);
351
352 // If left is true, skip right evaluation and keep true on stack
353 int end_jump = codegen_emit_jump(codegen, OPCODE_JUMP_IF_TRUE);
354
355 // Left was false, pop it and evaluate right
356 codegen_emit_byte(codegen, OPCODE_POP);
357 codegen_node(codegen, binop->right);
358
359 // Patch the jump to here (if left was true, we skip right and keep left's true value)
360 codegen_patch_jump(codegen, end_jump);
361 return;
362 }
363
364 // For all other operators, evaluate both operands
365 codegen_node(codegen, binop->left);
366 codegen_node(codegen, binop->right);
367
368 // Emit operation
369 if (strcmp(binop->op, "+") == 0) {
370 codegen_emit_byte(codegen, OPCODE_ADD);
371 } else if (strcmp(binop->op, "-") == 0) {
372 codegen_emit_byte(codegen, OPCODE_SUBTRACT);
373 } else if (strcmp(binop->op, "*") == 0) {
374 codegen_emit_byte(codegen, OPCODE_MULTIPLY);
375 } else if (strcmp(binop->op, "/") == 0) {
376 codegen_emit_byte(codegen, OPCODE_DIVIDE);
377 } else if (strcmp(binop->op, "%") == 0) {
378 codegen_emit_byte(codegen, OPCODE_MODULO);
379 } else if (strcmp(binop->op, "==") == 0) {
380 codegen_emit_byte(codegen, OPCODE_EQUAL);
381 } else if (strcmp(binop->op, "!=") == 0) {
382 codegen_emit_byte(codegen, OPCODE_NOT_EQUAL);
383 } else if (strcmp(binop->op, ">") == 0) {
384 codegen_emit_byte(codegen, OPCODE_GREATER);
385 } else if (strcmp(binop->op, ">=") == 0) {
386 codegen_emit_byte(codegen, OPCODE_GREATER_EQUAL);
387 } else if (strcmp(binop->op, "<") == 0) {
388 codegen_emit_byte(codegen, OPCODE_LESS);
389 } else if (strcmp(binop->op, "<=") == 0) {
390 codegen_emit_byte(codegen, OPCODE_LESS_EQUAL);
391 } else {
392 codegen_error(codegen, "Unknown binary operator");
393 }
394}

473 // Add more cases as needed

497 // Create main function context

515// Optimization: constant folding
516void codegen_constant_folding(Chunk* chunk) {
517 // Simple peephole optimization for constant operations
518 for (int i = 0; i < chunk->count - 2; i++) {
519 if (chunk->code[i] == OPCODE_CONSTANT &&
520 chunk->code[i + 2] == OPCODE_CONSTANT) {
521 // Two consecutive constants
522 uint8_t const1_idx = chunk->code[i + 1];
523 uint8_t const2_idx = chunk->code[i + 3];
524
525 if (i + 4 < chunk->count) {
526 uint8_t op = chunk->code[i + 4];
527 double val1 = chunk->constants.data[const1_idx];
528 double val2 = chunk->constants.data[const2_idx];
529 double result = 0;
530 bool can_fold = true;
531
532 switch (op) {
533 case OPCODE_ADD: result = val1 + val2; break;
534 case OPCODE_SUBTRACT: result = val1 - val2; break;
535 case OPCODE_MULTIPLY: result = val1 * val2; break;
536 case OPCODE_DIVIDE:
537 if (val2 != 0) result = val1 / val2;
538 else can_fold = false;
539 break;
540 default: can_fold = false; break;
541 }
542
543 if (can_fold) {
544 // Replace with single constant
545 int new_const = chunk_add_constant(chunk, result);
546 chunk->code[i + 1] = (uint8_t)new_const;
547 // Mark removed instructions as NOPs
548 chunk->code[i + 2] = OPCODE_NOP;
549 chunk->code[i + 3] = OPCODE_NOP;
550 chunk->code[i + 4] = OPCODE_NOP;
551 }
552 }
553 }
554 }

557// Optimization: dead code elimination
558void codegen_dead_code_elimination(Chunk* chunk) {
559 // Remove unreachable code after RETURN
560 for (int i = 0; i < chunk->count; i++) {
561 if (chunk->code[i] == OPCODE_RETURN) {
562 // Mark everything after as NOP until next label/jump target
563 for (int j = i + 1; j < chunk->count; j++) {
564 uint8_t op = chunk->code[j];
565 if (op == OPCODE_JUMP || op == OPCODE_JUMP_IF_FALSE ||
566 op == OPCODE_JUMP_IF_TRUE || op == OPCODE_LOOP) {
567 break; // Stop at jump targets
568 }
569 chunk->code[j] = OPCODE_NOP;
570 }
571 }
572 }

575// Optimization: peephole optimization
576void codegen_peephole_optimization(Chunk* chunk) {
577 // Need at least 2 bytes for DUP+POP pattern, 3 bytes for CONSTANT+arg+POP pattern
578 for (int i = 0; i < chunk->count - 1; i++) {
579 // CONSTANT followed by POP -> remove both
580 // Need to check i+2 is valid before accessing it
581 if (i + 2 < chunk->count &&
582 chunk->code[i] == OPCODE_CONSTANT &&
583 chunk->code[i + 2] == OPCODE_POP) {
584 chunk->code[i] = OPCODE_NOP;
585 chunk->code[i + 1] = OPCODE_NOP;
586 chunk->code[i + 2] = OPCODE_NOP;
587 }
588
589 // DUP followed by POP -> NOP
590 if (chunk->code[i] == OPCODE_DUP && chunk->code[i + 1] == OPCODE_POP) {
591 chunk->code[i] = OPCODE_NOP;
592 chunk->code[i + 1] = OPCODE_NOP;
593 }
594 }
595}

613// Disassemble a single instruction
614int codegen_disassemble_instruction(const Chunk* chunk, int offset) {
615 printf("%04d ", offset);
616
617 uint8_t instruction = chunk->code[offset];
618
619 if (instruction >= OPCODE_COUNT) {
620 printf("Unknown opcode %d\n", instruction);
621 return offset + 1;
622 }
623
624 const char* name = opcode_names[instruction];
625
626 switch (instruction) {
627 case OPCODE_CONSTANT: {
628 uint8_t constant = chunk->code[offset + 1];
629 printf("%-16s %4d '", name, constant);
630 printf("%g", chunk->constants.data[constant]);
631 printf("'\n");
632 return offset + 2;
633 }
634
635 case OPCODE_GET_LOCAL:
636 case OPCODE_SET_LOCAL:
637 case OPCODE_GET_GLOBAL:
638 case OPCODE_SET_GLOBAL:
639 case OPCODE_DEFINE_GLOBAL:
640 case OPCODE_CALL: {
641 uint8_t slot = chunk->code[offset + 1];
642 printf("%-16s %4d\n", name, slot);
643 return offset + 2;
644 }
645
646 case OPCODE_JUMP:
647 case OPCODE_JUMP_IF_FALSE:
648 case OPCODE_JUMP_IF_TRUE:
649 case OPCODE_LOOP: {
650 uint16_t jump = (uint16_t)(chunk->code[offset + 1] << 8);
651 jump |= chunk->code[offset + 2];
652 printf("%-16s %4d -> %d\n", name, offset,
653 instruction == OPCODE_LOOP ? offset - jump : offset + 3 + jump);
654 return offset + 3;
655 }
656
657 default:
658 printf("%s\n", name);
659 return offset + 1;
660 }
661}
