50void adam_step(AdamOptimizer* opt, double* params, double* gradients) {
51 if (!opt || !params || !gradients) return;
52
53 opt->step_count++;
54
55 // Bias correction terms
56 double bias_correction1 = 1.0 - pow(opt->config.beta1, opt->step_count);
57 double bias_correction2 = 1.0 - pow(opt->config.beta2, opt->step_count);
58
59 for (size_t i = 0; i < opt->num_params; i++) {
60 double grad = gradients[i];
61
62 // Apply weight decay
63 if (opt->config.weight_decay > 0.0) {
64 grad += opt->config.weight_decay * params[i];
65 }
66
67 // Update biased first moment estimate
68 opt->m[i] = opt->config.beta1 * opt->m[i] + (1.0 - opt->config.beta1) * grad;
69
70 // Update biased second raw moment estimate
71 opt->v[i] = opt->config.beta2 * opt->v[i] + (1.0 - opt->config.beta2) * grad * grad;
72
73 // Compute bias-corrected first moment estimate
74 double m_hat = opt->m[i] / bias_correction1;
75
76 // Compute bias-corrected second raw moment estimate
77 double v_hat = opt->v[i] / bias_correction2;
78
79 // AMSGrad variant
80 if (opt->config.amsgrad) {
81 opt->v_max[i] = fmax(opt->v_max[i], v_hat);
82 v_hat = opt->v_max[i];
83 }
84
85 // Update parameters
86 params[i] -= opt->config.learning_rate * m_hat / (sqrt(v_hat) + opt->config.epsilon);
87 }
88}
