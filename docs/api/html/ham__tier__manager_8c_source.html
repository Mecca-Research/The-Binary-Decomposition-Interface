63bool should_promote(const HamRegion* region, const AccessPattern* pattern, const HamPolicy* policy) {
64 if (!region || !pattern || !policy) return false;
65
66 // Promote if:
67 // 1. Access frequency is high (low entropy in access pattern)
68 // 2. Current tier is not already CRITICAL
69
70 if (region->tier == HAM_CRITICAL) return false;
71
72 float access_entropy = pattern->entropy_score;
73
74 // Low entropy = regular access pattern = hot data
75 if (access_entropy < policy->promotion_threshold) {
76 return true;
77 }
78
79 // High access count
80 if (region->stats.access_count > 1000) {
81 return true;
82 }
83
84 return false;
85}

112int ham_auto_promote(HamTierManager* manager, RegionId region_id) {

133 break;

139 break;
140 case HAM_CRITICAL:

184int ham_tier_manager_update(HamTierManager* manager, uint64_t current_cycle) {
185 if (!manager) return -1;
186
187 // Update all regions
188 for (size_t i = 0; i < manager->region_count; i++) {
189 HamRegion* region = manager->regions[i];
190 AccessPattern* pattern = manager->access_patterns[i];
191
192 // Record access if recent
193 if (region->stats.last_access_cycle + manager->policy.access_window >= current_cycle) {
194 access_pattern_record(pattern, region->stats.last_access_cycle);
195 }
196
197 // Check for promotion
198 if (should_promote(region, pattern, &manager->policy)) {
199 ham_auto_promote(manager, region->id);
200 }
201
202 // Check for demotion
203 if (should_demote(region, pattern, &manager->policy)) {
204 ham_auto_demote(manager, region->id);
205 }
206 }
207
208 return 0;
