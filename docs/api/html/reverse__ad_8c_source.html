8GradientTape* tape_create(size_t initial_capacity) {
9 GradientTape* tape = malloc(sizeof(GradientTape));
10 if (!tape) return NULL;
11
12 tape->entries = malloc(initial_capacity * sizeof(TapeEntry));
13 if (!tape->entries) {
14 free(tape);
15 return NULL;
16 }
17
18 tape->gradients = calloc(initial_capacity, sizeof(double));
19 if (!tape->gradients) {
20 free(tape->entries);
21 free(tape);
22 return NULL;
23 }
24
25 tape->size = 0;
26 tape->capacity = initial_capacity;
27 tape->num_variables = 0;
28 tape->grad_capacity = initial_capacity;
29
30 return tape;
31}

49int tape_resize(GradientTape* tape, size_t new_capacity) {
50 if (tape->capacity >= new_capacity) {
51 return 0;
52 }
53
54 // Perform both reallocations first, storing results in temporary variables
55 TapeEntry* new_entries = realloc(tape->entries, new_capacity * sizeof(TapeEntry));
56 if (!new_entries) {
57 fprintf(stderr, "Error: Failed to reallocate tape entries buffer\n");
58 return -1; // First realloc failed, no state changed
59 }
60
61 double* new_grads = NULL;
62 if (new_capacity > tape->grad_capacity) {
63 new_grads = realloc(tape->gradients, new_capacity * sizeof(double));
64 if (!new_grads) {
65 fprintf(stderr, "Error: Failed to reallocate tape gradients buffer\n");
66 // Second realloc failed - restore entries to prevent memory leak
67 // Keep the successfully reallocated entries buffer but don't update capacity
68 tape->entries = new_entries;
69 return -1; // Return failure, tape remains in consistent state
70 }
71 }
72
73 // Both reallocations succeeded - now atomically update all state
74 tape->entries = new_entries;
75 tape->capacity = new_capacity;
76
77 if (new_grads) {
78 memset(new_grads + tape->grad_capacity, 0,
79 (new_capacity - tape->grad_capacity) * sizeof(double));
80 tape->gradients = new_grads;
81 tape->grad_capacity = new_capacity;
82 }
83
84 return 0;
85}

87static size_t tape_add_entry(GradientTape* tape, TapeEntry entry) {
88 if (tape->size >= tape->capacity) {
89 if (tape_resize(tape, tape->capacity * 2) != 0) {
90 fprintf(stderr, "Error: Failed to resize tape, cannot add entry\n");
91 return (size_t)-1; // Return error indicator
92 }
93 }
94
95 size_t id = tape->size;
96 tape->entries[tape->size++] = entry;
97 return id;
98}

300void tape_backward(GradientTape* tape, size_t output_id) {
301 if (!tape || output_id >= tape->size) return;
302
303 // Initialize all gradients to zero
304 memset(tape->gradients, 0, tape->grad_capacity * sizeof(double));
305
306 // Set output gradient to 1.0
307 tape->gradients[output_id] = 1.0;
308
309 // Backward pass through tape in reverse order
310 for (size_t i = tape->size; i > 0; i--) {
311 size_t idx = i - 1;
312 TapeEntry* entry = &tape->entries[idx];
313 double grad_out = tape->gradients[entry->output_id];
314
315 if (grad_out == 0.0) continue;
316
317 switch (entry->op) {
318 case OP_ADD:
319 tape->gradients[entry->input_ids[0]] += grad_out;
320 tape->gradients[entry->input_ids[1]] += grad_out;
321 break;
322
323 case OP_SUB:
324 tape->gradients[entry->input_ids[0]] += grad_out;
325 tape->gradients[entry->input_ids[1]] -= grad_out;
326 break;
327
328 case OP_MUL:
329 tape->gradients[entry->input_ids[0]] += grad_out * entry->values[2];
330 tape->gradients[entry->input_ids[1]] += grad_out * entry->values[1];
331 break;
332
333 case OP_DIV:
334 tape->gradients[entry->input_ids[0]] += grad_out / entry->values[2];
335 tape->gradients[entry->input_ids[1]] -= grad_out * entry->values[1] /
336 (entry->values[2] * entry->values[2]);
337 break;
338
339 case OP_NEG:
340 tape->gradients[entry->input_ids[0]] -= grad_out;
341 break;
342
343 case OP_SIN:
344 tape->gradients[entry->input_ids[0]] += grad_out * cos(entry->values[1]);
345 break;
346
347 case OP_COS:
348 tape->gradients[entry->input_ids[0]] -= grad_out * sin(entry->values[1]);
349 break;
350
351 case OP_TAN: {
352 double cos_val = cos(entry->values[1]);
353 tape->gradients[entry->input_ids[0]] += grad_out / (cos_val * cos_val);
354 break;
355 }
356
357 case OP_EXP:
358 tape->gradients[entry->input_ids[0]] += grad_out * entry->values[0];
359 break;
360
361 case OP_LOG:
362 tape->gradients[entry->input_ids[0]] += grad_out / entry->values[1];
363 break;
364
365 case OP_SQRT:
366 tape->gradients[entry->input_ids[0]] += grad_out / (2.0 * entry->values[0]);
367 break;
368
369 case OP_POW:
370 tape->gradients[entry->input_ids[0]] += grad_out * entry->param *
371 pow(entry->values[1], entry->param - 1.0);
372 break;
373
374 case OP_ABS:
375 tape->gradients[entry->input_ids[0]] += grad_out *
376 (entry->values[1] >= 0.0 ? 1.0 : -1.0);
377 break;
378
379 case OP_TANH: {
380 double tanh_val = entry->values[0];
381 tape->gradients[entry->input_ids[0]] += grad_out * (1.0 - tanh_val * tanh_val);
382 break;
383 }
384
385 case OP_SIGMOID: {
386 double sig = entry->values[0];
387 tape->gradients[entry->input_ids[0]] += grad_out * sig * (1.0 - sig);
388 break;
389 }
390
391 case OP_RELU:
392 tape->gradients[entry->input_ids[0]] += grad_out * (entry->values[1] > 0.0 ? 1.0 : 0.0);
393 break;
394
395 case OP_CONSTANT:
396 case OP_VARIABLE:
397 // No backward pass needed
398 break;
399 }
400 }
