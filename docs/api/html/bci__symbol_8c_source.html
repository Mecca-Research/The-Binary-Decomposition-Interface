37bool symbol_table_add(SymbolTable* table, Symbol symbol) {
38 // Check for redeclaration in the same scope.
39 for (int i = table->symbols.len - 1; i >= 0; i--) {
40 Symbol* existing = &table->symbols.data[i];
41 if (existing->depth < table->scope_depth) {
42 break; // We've left the current scope.
43 }
44 if (strncmp(existing->name.start, symbol.name.start, symbol.name.length) == 0 &&
45 strlen(existing->name.start) == symbol.name.length) {
46 // Error: Symbol with same name already declared in this scope.
47 return false;
48 }
49 }
50
51 symbol.depth = table->scope_depth;
52 bci_vec_push(&table->symbols, symbol);
53 return true;
54}

56Symbol* symbol_table_lookup(SymbolTable* table, Token name) {
57 // Search backwards to find the innermost declaration (shadowing).
58 for (int i = table->symbols.len - 1; i >= 0; i--) {
59 Symbol* symbol = &table->symbols.data[i];
60 if (name.length == symbol->name.length &&
61 memcmp(name.start, symbol->name.start, name.length) == 0) {
62 return symbol;
63 }
64 }
65 return nullptr;
66}
