22// Recursively frees an AST node and all its children.
23void ast_free_node(AstNode* node) {
24 if (!node) return;
25
26 switch (node->kind) {
27 case AST_NODE_BINARY_OP:
28 ast_free_node(node->as.binary_op.left);
29 ast_free_node(node->as.binary_op.right);
30 break;
31 case AST_NODE_BLOCK:
32 case AST_NODE_PROGRAM:
33 if (node->as.block) {
34 for (size_t i = 0; i < node->as.block->len; ++i) {
35 ast_free_node(node->as.block->data[i]);
36 }
37 bci_vec_free(node->as.block);
38 free(node->as.block);
39 }
40 break;
41 case AST_NODE_VAR_DECL:
42 ast_free_node(node->as.var_decl.initializer);
43 break;
44 case AST_NODE_FUNC_DECL:
45 if (node->as.func_decl.params.data) {
46 // Params are AstVarDecl, need careful freeing if they are complex
47 for (size_t i = 0; i < node->as.func_decl.params.len; ++i) {
48 // Assuming params are simple and don't need deep free here
49 free(node->as.func_decl.params.data[i]);
50 }
51 bci_vec_free(&node->as.func_decl.params);
52 }
53 ast_free_node((AstNode*)node->as.func_decl.body); // Cast AstBlock back
54 break;
55 // Add cases for other node kinds that have heap-allocated children
56 case AST_NODE_LITERAL: // No children to free
57 case AST_NODE_VARIABLE: // No children to free
58 default:
59 break;
60 }
61
62 free(node);
63}
