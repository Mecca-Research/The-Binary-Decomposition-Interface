37void lr_scheduler_step(LRScheduler* scheduler) {
38 if (!scheduler) return;
39
40 scheduler->current_step++;
41
42 switch (scheduler->config.type) {
43 case LR_CONSTANT:
44 scheduler->current_lr = scheduler->config.initial_lr;
45 break;
46
47 case LR_STEP: {
48 size_t num_drops = scheduler->current_step / scheduler->config.step_size;
49 scheduler->current_lr = scheduler->config.initial_lr *
50 pow(scheduler->config.gamma, num_drops);
51 break;
52 }
53
54 case LR_EXPONENTIAL:
55 scheduler->current_lr = scheduler->config.initial_lr *
56 pow(scheduler->config.decay_rate, scheduler->current_step);
57 break;
58
59 case LR_COSINE: {
60 double progress = (double)scheduler->current_step / scheduler->config.T_max;
61 progress = fmin(progress, 1.0);
62 scheduler->current_lr = scheduler->config.min_lr +
63 (scheduler->config.initial_lr - scheduler->config.min_lr) *
64 0.5 * (1.0 + cos(M_PI * progress));
65 break;
66 }
67
68 case LR_COSINE_WARM_RESTARTS: {
69 size_t T_cur = scheduler->current_step;
70 size_t T_i = scheduler->config.T_0;
71 size_t restart_count = 0;
72
73 while (T_cur >= T_i) {
74 T_cur -= T_i;
75 T_i *= scheduler->config.T_mult;
76 restart_count++;
77 }
78
79 double progress = (double)T_cur / T_i;
80 scheduler->current_lr = scheduler->config.min_lr +
81 (scheduler->config.initial_lr - scheduler->config.min_lr) *
82 0.5 * (1.0 + cos(M_PI * progress));
83 break;
84 }
85
86 case LR_LINEAR_WARMUP:
87 if (scheduler->current_step < scheduler->config.warmup_steps) {
88 double progress = (double)scheduler->current_step / scheduler->config.warmup_steps;
89 scheduler->current_lr = scheduler->config.warmup_start_lr +
90 (scheduler->config.initial_lr - scheduler->config.warmup_start_lr) *
91 progress;
92 } else {
93 scheduler->current_lr = scheduler->config.initial_lr;
94 }
95 break;
96 }
97}
