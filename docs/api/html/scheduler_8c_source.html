36// The core logic of M4: The Policy Gate.
37static bool policy_gate_check(Scheduler* sched, GraphNode* node) {
38 // If not in secure mode, all checks pass.
39 if (!sched->policy.secure_mode) {
40 return true;
41 }
42
43 // In secure mode, the node must have attached metadata.
44 if (node->meta_off == 0 && sched->graph->meta_size > 0) { // meta_off is 0-based index
45 // Special case: if meta_off is 0 but points to valid meta, it's okay.
46 // A better system would use a sentinel value like SIZE_MAX.
47 // For now, we assume meta_off > 0 is required for attached meta.
48 } else if (node->meta_off > sched->graph->meta_size) {
49 printf("POLICY_GATE: REJECT Node %llu - No metadata in secure mode.\n", (unsigned long long)node->id);
50 return false;
51 }
52
53 NodeMeta* meta = (NodeMeta*)(sched->graph->meta_arena + node->meta_off);
54
55 // Check 1: The node must meet the minimum required proof class.
56 if ((meta->proof_class & sched->policy.required_proof_class) != sched->policy.required_proof_class) {
57 printf("POLICY_GATE: REJECT Node %llu - Does not meet required proof class.\n", (unsigned long long)node->id);
58 return false;
59 }
60
61 // Check 2: The metadata hash must be valid (Merkle-hashing).
62 uint8_t computed_hash[32];
63 aeon_hash_meta(node, computed_hash);
64 if (memcmp(meta->hash, computed_hash, 32) != 0) {
65 printf("POLICY_GATE: REJECT Node %llu - Metadata hash mismatch! (Tampering detected)\n", (unsigned long long)node->id);
66 return false;
67 }
68
69 printf("POLICY_GATE: PASS Node %llu.\n", (unsigned long long)node->id);
70 return true; // All checks passed.
71}

73int aeon_scheduler_run_wave(Scheduler* sched) {
74 if (!sched) return -1;
75 BdiGraph* g = sched->graph;
76
77 // 1. Find ready nodes (simple version: all nodes with no inputs).
78 // A real scheduler would track dependencies.
79 for (size_t i = 0; i < g->node_count; i++) {
80 if (g->nodes[i].input_count == 0) {
81 // In a real scheduler, we would manage the ready_set array.
82 GraphNode* node = &g->nodes[i];
83
84 // 2. Apply the Policy Gate.
85 if (policy_gate_check(sched, node)) {
86 // 3. Dispatch the node if it passes.
87 DeviceId hint = node->device_hint > 0 ? node->device_hint : DEVICE_ID_CPU;
88 DeviceVTable* device = sched->devices[hint];
89 if (device) {
90 printf("SCHEDULER: Dispatching Node %llu to Device '%s'.\n", (unsigned long long)node->id, device->name);
91 void* kernel;
92 device->lower(node, &kernel);
93 device->enqueue(kernel, nullptr, 0);
94 }
95 }
96 }
97 }
98
99 // 4. Sync devices.
100 for (size_t i = 1; i <= sched->device_count; i++) {
101 if (sched->devices[i]) sched->devices[i]->sync();
102 }
103
104 return 0;
105}
