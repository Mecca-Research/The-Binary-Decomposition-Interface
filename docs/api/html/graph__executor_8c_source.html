56bool graph_scheduler_schedule(GraphScheduler* scheduler) {
57 if (!scheduler || !scheduler->graph) return false;
58
59 uint64_t start_time = get_time_ns();
60
61 Graph* graph = scheduler->graph;
62
63 // Reset scheduler state
64 scheduler->ready_queue_head = 0;
65 scheduler->ready_queue_tail = 0;
66 scheduler->execution_order_count = 0;
67
68 // Initialize dependency counts
69 for (uint32_t i = 0; i < graph->node_count; i++) {
70 GraphNode* node = graph->nodes[i];
71 scheduler->remaining_deps[i] = node->dependency_count;
72
73 // Nodes with no dependencies are ready
74 if (node->dependency_count == 0) {
75 scheduler->ready_queue[scheduler->ready_queue_tail++] = i;
76 }
77 }
78
79 // Process ready queue
80 while (scheduler->ready_queue_head < scheduler->ready_queue_tail) {
81 uint32_t node_idx = scheduler->ready_queue[scheduler->ready_queue_head++];
82 GraphNode* node = graph->nodes[node_idx];
83
84 // Add to execution order
85 scheduler->execution_order[scheduler->execution_order_count++] = node->id;
86
87 // Update dependencies of successor nodes
88 for (uint32_t i = 0; i < node->output_count; i++) {
89 GraphEdge* edge = node->outputs[i];
90 if (!edge) continue;
91
92 // Find target node index
93 uint32_t target_idx = UINT32_MAX;
94 for (uint32_t j = 0; j < graph->node_count; j++) {
95 if (graph->nodes[j] == edge->target) {
96 target_idx = j;
97 break;
98 }
99 }
100
101 if (target_idx != UINT32_MAX) {
102 scheduler->remaining_deps[target_idx]--;
103 if (scheduler->remaining_deps[target_idx] == 0) {
104 scheduler->ready_queue[scheduler->ready_queue_tail++] = target_idx;
105 }
106 }
107 }
108 }
109
110 scheduler->scheduling_time_ns += get_time_ns() - start_time;
111 scheduler->total_scheduled++;
112
113 // Check if all nodes were scheduled (no cycles)
114 return scheduler->execution_order_count == graph->node_count;
115}

126void graph_scheduler_mark_completed(GraphScheduler* scheduler, uint32_t node_id) {
127 if (!scheduler) return;
128
129 // Find the node and update dependencies of its successors
130 GraphNode* node = graph_get_node(scheduler->graph, node_id);
131 if (!node) return;
132
133 for (uint32_t i = 0; i < node->output_count; i++) {
134 GraphEdge* edge = node->outputs[i];
135 if (!edge) continue;
136
137 // Find target node index
138 uint32_t target_idx = UINT32_MAX;
139 for (uint32_t j = 0; j < scheduler->graph->node_count; j++) {
140 if (scheduler->graph->nodes[j] == edge->target) {
141 target_idx = j;
142 break;
143 }
144 }
145
146 if (target_idx != UINT32_MAX) {
147 scheduler->remaining_deps[target_idx]--;
148 if (scheduler->remaining_deps[target_idx] == 0) {
149 // Add to ready queue if there's space
150 if (scheduler->ready_queue_tail < scheduler->ready_queue_size) {
151 scheduler->ready_queue[scheduler->ready_queue_tail++] = target_idx;
152 }
153 }
154 }
155 }

180GraphExecutionContext* graph_execution_context_create(Graph* graph) {
181 if (!graph) return NULL;
182
183 GraphExecutionContext* context = (GraphExecutionContext*)calloc(1, sizeof(GraphExecutionContext));
184 if (!context) return NULL;
185
186 context->graph = graph; // Store reference to the graph
187 context->input_count = graph->input_count;
188 context->output_count = graph->output_count;
189
190 if (context->input_count > 0) {
191 context->input_values = (GraphValue*)calloc(context->input_count, sizeof(GraphValue));
192 }
193
194 if (context->output_count > 0) {
195 context->output_values = (GraphValue*)calloc(context->output_count, sizeof(GraphValue));
196 }
197
198 // Allocate execution state tracking
199 context->node_executed = (bool*)calloc(graph->node_count, sizeof(bool));
200 context->node_outputs = (GraphValue**)calloc(graph->node_count, sizeof(GraphValue*));
201
202 // Allocate output arrays for each node
203 for (uint32_t i = 0; i < graph->node_count; i++) {
204 GraphNode* node = graph->nodes[i];
205 if (node->max_outputs > 0) {
206 context->node_outputs[i] = (GraphValue*)calloc(node->max_outputs, sizeof(GraphValue));
207 }
208 }
209
210 // Initialize memory pool
211 context->memory_limit = 1024 * 1024; // 1MB default
212 context->memory_pool = malloc(context->memory_limit);
213 context->memory_used = 0;
214
215 if (!context->node_executed || !context->node_outputs || !context->memory_pool) {
216 graph_execution_context_destroy(context);
217 return NULL;
218 }
219
220 return context;
221}

281GraphExecutionResult graph_executor_execute(GraphExecutor* executor, Graph* graph,
282 GraphValue* inputs, uint32_t input_count) {
283 GraphExecutionResult result = {0};
284
285 if (!executor || !graph) {
286 result.error_message = strdup("Invalid executor or graph");
287 return result;
288 }
289
290 uint64_t start_time = get_time_ns();
291
292 // Create execution context
293 GraphExecutionContext* context = graph_execution_context_create(graph);
294 if (!context) {
295 result.error_message = strdup("Failed to create execution context");
296 return result;
297 }
298
299 // Set input values
300 if (inputs && input_count > 0) {
301 uint32_t copy_count = input_count < context->input_count ? input_count : context->input_count;
302 memcpy(context->input_values, inputs, copy_count * sizeof(GraphValue));
303 }
304
305 // Execute the graph
306 bool success = false;
307
308 switch (executor->config.mode) {
309 case GRAPH_EXEC_SEQUENTIAL:
310 success = graph_executor_execute_sequential(executor, graph, context);
311 break;
312 case GRAPH_EXEC_PARALLEL:
313 success = graph_executor_execute_parallel(executor, graph, context);
314 break;
315 case GRAPH_EXEC_JIT_COMPILED:
316 if (executor->jit_vm) {
317 success = graph_executor_execute_compiled(executor, graph,
318 inputs, input_count,
319 context->output_values, &context->output_count);
320 } else {
321 success = graph_executor_execute_sequential(executor, graph, context);
322 }
323 break;
324 default:
325 success = graph_executor_execute_sequential(executor, graph, context);
326 break;
327 }
328
329 result.success = success;
330 result.execution_time_ns = get_time_ns() - start_time;
331 result.nodes_executed = context->nodes_executed;
332
333 if (success) {
334 // Copy output values
335 result.output_count = context->output_count;
336 if (result.output_count > 0) {
337 result.output_values = (GraphValue*)malloc(result.output_count * sizeof(GraphValue));
338 if (result.output_values) {
339 memcpy(result.output_values, context->output_values,
340 result.output_count * sizeof(GraphValue));
341 }
342 }
343
344 executor->successful_executions++;
345 } else {
346 result.error_message = context->error_message ? strdup(context->error_message) :
347 strdup("Graph execution failed");
348 executor->failed_executions++;
349 }
350
351 // Update statistics
352 executor->total_executions++;
353 executor->total_execution_time_ns += result.execution_time_ns;
354 executor->nodes_executed += result.nodes_executed;
355
356 graph_execution_context_destroy(context);
357 return result;
358}

360// Sequential execution implementation
361static bool graph_executor_execute_sequential(GraphExecutor* executor, Graph* graph,
362 GraphExecutionContext* context) {
363 if (!executor || !graph || !context) return false;
364
365 // Create scheduler if needed
366 if (!executor->scheduler) {
367 executor->scheduler = graph_scheduler_create(graph);
368 if (!executor->scheduler) return false;
369 }
370
371 // Schedule the graph
372 if (!graph_scheduler_schedule(executor->scheduler)) {
373 graph_execution_context_set_error(context, "Failed to schedule graph (cycle detected)", 0);
374 return false;
375 }
376
377 // Execute nodes in scheduled order
378 for (uint32_t i = 0; i < executor->scheduler->execution_order_count; i++) {
379 uint32_t node_id = executor->scheduler->execution_order[i];
380 GraphNode* node = graph_get_node(graph, node_id);
381
382 if (!node) {
383 graph_execution_context_set_error(context, "Invalid node in execution order", node_id);
384 return false;
385 }
386
387 // Prepare input values for this node
388 GraphValue* node_inputs = NULL;
389 if (node->input_count > 0) {
390 node_inputs = (GraphValue*)malloc(node->input_count * sizeof(GraphValue));
391 if (!node_inputs) {
392 graph_execution_context_set_error(context, "Memory allocation failed", node_id);
393 return false;
394 }
395
396 // Collect inputs from predecessor nodes or graph inputs
397 for (uint32_t j = 0; j < node->input_count; j++) {
398 GraphEdge* edge = node->inputs[j];
399 if (edge && edge->source) {
400 // Find source node index
401 uint32_t source_idx = UINT32_MAX;
402 for (uint32_t k = 0; k < graph->node_count; k++) {
403 if (graph->nodes[k] == edge->source) {
404 source_idx = k;
405 break;
406 }
407 }
408
409 if (source_idx != UINT32_MAX && context->node_outputs[source_idx]) {
410 node_inputs[j] = context->node_outputs[source_idx][edge->source_output];
411 } else if (edge->source->type == GRAPH_NODE_INPUT) {
412 // Input node - get value from context inputs
413 for (uint32_t k = 0; k < graph->input_count; k++) {
414 if (graph->inputs[k] == edge->source) {
415 if (k < context->input_count) {
416 node_inputs[j] = context->input_values[k];
417 }
418 break;
419 }
420 }
421 }
422 } else {
423 // No input edge - use zero value
424 node_inputs[j] = graph_value_zero(GRAPH_TYPE_F64);
425 }
426 }
427 }
428
429 // Find node index for output storage
430 uint32_t node_idx = UINT32_MAX;
431 for (uint32_t j = 0; j < graph->node_count; j++) {
432 if (graph->nodes[j] == node) {
433 node_idx = j;
434 break;
435 }
436 }
437
438 if (node_idx == UINT32_MAX) {
439 free(node_inputs);
440 graph_execution_context_set_error(context, "Node index not found", node_id);
441 return false;
442 }
443
444 // Execute the node
445 bool node_success = graph_execute_node(node, node_inputs,
446 context->node_outputs[node_idx], context);
447
448 free(node_inputs);
449
450 if (!node_success) {
451 graph_execution_context_set_error(context, "Node execution failed", node_id);
452 return false;
453 }
454
455 context->node_executed[node_idx] = true;
456 context->nodes_executed++;
457
458 // If this is an output node, copy to context outputs
459 if (node->type == GRAPH_NODE_OUTPUT) {
460 for (uint32_t j = 0; j < graph->output_count; j++) {
461 if (graph->outputs[j] == node && j < context->output_count) {
462 context->output_values[j] = context->node_outputs[node_idx][0];
463 break;
464 }
465 }
466 }
467 }
468
469 return true;
470}

472// Node execution implementation
473bool graph_execute_node(GraphNode* node, GraphValue* inputs, GraphValue* outputs,
474 GraphExecutionContext* context) {
475 if (!node || !outputs) return false;
476
477 uint64_t start_time = get_time_ns();
478 bool success = false;
479
480 switch (node->type) {
481 case GRAPH_NODE_CONSTANT:
482 success = graph_execute_constant_node(node, outputs);
483 break;
484
485 case GRAPH_NODE_INPUT:
486 // Input nodes get their values from the execution context
487 if (context && context->input_values) {
488 // Find the input index for this node by searching in the graph's inputs array
489 uint32_t input_index = UINT32_MAX;
490 Graph* graph = context->graph;
491 if (graph) {
492 for (uint32_t i = 0; i < graph->input_count; i++) {
493 if (graph->inputs[i] == node) {
494 input_index = i;
495 break;
496 }
497 }
498 }
499
500 // Use the actual input value if found and within bounds
501 if (input_index != UINT32_MAX && input_index < context->input_count) {
502 outputs[0] = context->input_values[input_index];
503 success = true;
504 } else {
505 // Error case: input node not found or index out of bounds
506 outputs[0] = graph_value_zero(GRAPH_TYPE_F64);
507 success = false;
508 }
509 } else {
510 // No context or input values provided
511 outputs[0] = graph_value_zero(GRAPH_TYPE_F64);
512 success = false;
513 }
514 break;
515
516 case GRAPH_NODE_OUTPUT:
517 // Output nodes just pass through their input
518 if (inputs) {
519 outputs[0] = inputs[0];
520 } else {
521 outputs[0] = graph_value_zero(GRAPH_TYPE_F64);
522 }
523 success = true;
524 break;
525
526 case GRAPH_NODE_ADD:
527 case GRAPH_NODE_SUB:
528 case GRAPH_NODE_MUL:
529 case GRAPH_NODE_DIV:
530 case GRAPH_NODE_MOD:
531 success = graph_execute_arithmetic_node(node, inputs, outputs);
532 break;
533
534 case GRAPH_NODE_AND:
535 case GRAPH_NODE_OR:
536 case GRAPH_NODE_XOR:
537 case GRAPH_NODE_NOT:
538 success = graph_execute_logical_node(node, inputs, outputs);
539 break;
540
541 case GRAPH_NODE_EQ:
542 case GRAPH_NODE_NE:
543 case GRAPH_NODE_LT:
544 case GRAPH_NODE_LE:
545 case GRAPH_NODE_GT:
546 case GRAPH_NODE_GE:
547 success = graph_execute_comparison_node(node, inputs, outputs);
548 break;
549
550 case GRAPH_NODE_BRANCH:
551 case GRAPH_NODE_PHI:
552 case GRAPH_NODE_SELECT:
553 success = graph_execute_control_node(node, inputs, outputs, context);
554 break;
555
556 case GRAPH_NODE_LOAD:
557 case GRAPH_NODE_STORE:
558 case GRAPH_NODE_ALLOC:
559 success = graph_execute_memory_node(node, inputs, outputs, context);
560 break;
561
562 case GRAPH_NODE_CAST:
563 if (inputs && node->custom_data) {
564 struct {
565 GraphDataType from_type;
566 GraphDataType to_type;
567 } *cast_data = (void*)node->custom_data;
568 outputs[0] = graph_value_cast(&inputs[0], cast_data->to_type);
569 success = true;
570 }
571 break;
572
573 case GRAPH_NODE_CUSTOM:
574 success = graph_execute_custom_node(node, inputs, outputs, context);
575 break;
576
577 default:
578 success = false;
579 break;
580 }
581
582 uint64_t execution_time = get_time_ns() - start_time;
583 node->execution_time_ns += execution_time;
584 node->execution_count++;
585
586 return success;
587}

779 if (!executor || !stats) return;

836// Memory management
837void* graph_executor_alloc(GraphExecutionContext* context, size_t size) {
838 if (!context || !context->memory_pool) return malloc(size);
839
840 // Align to 8-byte boundary
841 size = (size + 7) & ~7;
842
843 if (context->memory_used + size > context->memory_limit) {
844 return NULL; // Out of memory
845 }
846
847 void* ptr = (char*)context->memory_pool + context->memory_used;
848 context->memory_used += size;
849 return ptr;
850}
