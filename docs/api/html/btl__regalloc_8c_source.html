34BTL_RegAllocator* btl_regalloc_create(size_t num_registers, BTL_RegisterType type) {
35 BTL_RegAllocator *allocator = malloc(sizeof(BTL_RegAllocator));
36 if (!allocator) return NULL;
37
38 allocator->registers = calloc(num_registers, sizeof(BTL_Register));
39 if (!allocator->registers) {
40 free(allocator);
41 return NULL;
42 }
43
44 allocator->num_registers = num_registers;
45 allocator->type = type;
46
47 // Initialize registers
48 for (size_t i = 0; i < num_registers; i++) {
49 allocator->registers[i].id = i;
50 allocator->registers[i].type = type;
51 allocator->registers[i].allocated = false;
52 allocator->registers[i].live_start = 0;
53 allocator->registers[i].live_end = 0;
54 allocator->registers[i].name = btl_regalloc_get_register_name(i, type);
55 }
56
57 // Initialize intervals
58 allocator->intervals_capacity = MAX_INTERVALS;
59 allocator->intervals = calloc(MAX_INTERVALS, sizeof(BTL_LiveInterval));
60 if (!allocator->intervals) {
61 free(allocator->registers);
62 free(allocator);
63 return NULL;
64 }
65 allocator->num_intervals = 0;
66
67 allocator->spill_count = 0;
68 allocator->next_spill_location = 0;
69
70 return allocator;
71}

80int btl_regalloc_acquire(BTL_RegAllocator *allocator, uint32_t var_id,
81 uint32_t start, uint32_t end) {
82 if (!allocator) return -1;
83
84 // Find a free register
85 for (size_t i = 0; i < allocator->num_registers; i++) {
86 if (!allocator->registers[i].allocated) {
87 allocator->registers[i].allocated = true;
88 allocator->registers[i].live_start = start;
89 allocator->registers[i].live_end = end;
90
91 // NOTE: Do NOT add interval here during linear scan.
92 // Intervals should already exist before the scan begins.
93 // Adding intervals during scan causes the loop to process
94 // newly appended entries, leading to infinite growth.
95
96 return (int)i;
97 }
98 }
99
100 return -1; // No free register
101}

142// Comparison function for sorting intervals by start point
143static int compare_intervals(const void *a, const void *b) {
144 const BTL_LiveInterval *ia = (const BTL_LiveInterval*)a;
145 const BTL_LiveInterval *ib = (const BTL_LiveInterval*)b;
146 return (int)ia->start - (int)ib->start;

149// Linear scan register allocation algorithm
150bool btl_regalloc_linear_scan(BTL_RegAllocator *allocator) {
151 if (!allocator || allocator->num_intervals == 0) return false;
152
153 // Sort intervals by start point
154 qsort(allocator->intervals, allocator->num_intervals,
155 sizeof(BTL_LiveInterval), compare_intervals);
156
157 // Active list (intervals currently using registers)
158 BTL_LiveInterval *active[256];
159 size_t active_count = 0;
160
161 // Process each interval
162 for (size_t i = 0; i < allocator->num_intervals; i++) {
163 BTL_LiveInterval *current = &allocator->intervals[i];
164
165 // Expire old intervals
166 for (size_t j = 0; j < active_count; ) {
167 if (active[j]->end <= current->start) {
168 // Release register
169 if (active[j]->assigned_register >= 0) {
170 btl_regalloc_release(allocator, active[j]->assigned_register);
171 }
172 // Remove from active list
173 active[j] = active[--active_count];
174 } else {
175 j++;
176 }
177 }
178
179 // Try to allocate a register
180 int reg = btl_regalloc_acquire(allocator, current->var_id,
181 current->start, current->end);
182
183 if (reg >= 0) {
184 current->assigned_register = reg;
185 active[active_count++] = current;
186 } else {
187 // Spill: no free register
188 current->spilled = true;
189 current->spill_location = allocator->next_spill_location++;
190 allocator->spill_count++;
191 }
192 }
193
194 return true;
195}

220const char* btl_regalloc_get_register_name(int reg_id, BTL_RegisterType type) {
221 if (reg_id < 0) return "INVALID";

223 switch (type) {

227 case BTL_REG_FLOAT:
228 case BTL_REG_VECTOR:

230 break;
231 default:
232 break;
233 }
