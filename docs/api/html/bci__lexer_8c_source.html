12// Checks if the lexer has reached the end of the source code.
13static bool is_at_end(Lexer* lexer) {
14 return *lexer->current == '\0';
15}
16
17// Consumes and returns the current character, advancing the lexer.
18static char advance(Lexer* lexer) {
19 lexer->current++;
20 return lexer->current[-1];
21}
22
23// Returns the current character without consuming it.
24static char peek(Lexer* lexer) {
25 return *lexer->current;
26}
27
28// Returns the next character without consuming it.
29static char peek_next(Lexer* lexer) {
30 if (is_at_end(lexer)) return '\0';
31 return lexer->current[1];
32}
33
34// If the current character matches 'expected', consumes it and returns true.
35static bool match(Lexer* lexer, char expected) {
36 if (is_at_end(lexer)) return false;
37 if (*lexer->current != expected) return false;
38 lexer->current++;
39 return true;
40}

42// Creates a token of the given kind.
43static Token make_token(Lexer* lexer, TokenKind kind) {
44 Token token;
45 token.kind = kind;
46 token.start = lexer->start;
47 token.length = (int)(lexer->current - lexer->start);
48 token.line = lexer->line;
49 return token;
50}

52// Creates an error token.
53static Token error_token(Lexer* lexer, const char* message) {
54 Token token;
55 token.kind = TOKEN_ERROR;

57 token.length = (int)strlen(message);

62// Skips whitespace and comments.
63static void skip_whitespace(Lexer* lexer) {
64 for (;;) {
65 char c = peek(lexer);
66 switch (c) {
67 case ' ':
68 case '\r':
69 case '\t':
70 advance(lexer);
71 break;
72 case '\n':
73 lexer->line++;
74 advance(lexer);
75 break;
76 case '/':
77 if (peek_next(lexer) == '/') {
78 // A comment goes until the end of the line.
79 while (peek(lexer) != '\n' && !is_at_end(lexer)) advance(lexer);
80 } else {
81 return;
82 }
83 break;
84 default:
85 return;
86 }
87 }
88}

90// Checks if an identifier is a keyword.
91static TokenKind check_keyword(Lexer* lexer, int start, int length, const char* rest, TokenKind kind) {
92 if (lexer->current - lexer->start == start + length &&
93 memcmp(lexer->start + start, rest, length) == 0) {
94 return kind;
95 }
96 return TOKEN_IDENTIFIER;
97}

99// Determines if an identifier is a keyword or a user-defined name.
100static TokenKind identifier_type(Lexer* lexer) {
101 switch (lexer->start[0]) {
102 case 'a': return check_keyword(lexer, 1, 2, "nd", TOKEN_AND);
103 case 'c': return check_keyword(lexer, 1, 4, "lass", TOKEN_CLASS);
104 case 'e': return check_keyword(lexer, 1, 3, "lse", TOKEN_ELSE);
105 case 'f':
106 if (lexer->current - lexer->start > 1) {
107 switch (lexer->start[1]) {
108 case 'a': return check_keyword(lexer, 2, 3, "lse", TOKEN_FALSE);
109 case 'o': return check_keyword(lexer, 2, 1, "r", TOKEN_FOR);
110 case 'u': return check_keyword(lexer, 2, 1, "n", TOKEN_FUN);
111 }
112 }
113 break;
114 case 'i': return check_keyword(lexer, 1, 1, "f", TOKEN_IF);
115 case 'n': return check_keyword(lexer, 1, 2, "il", TOKEN_NIL);
116 case 'o': return check_keyword(lexer, 1, 1, "r", TOKEN_OR);
117 case 'p': return check_keyword(lexer, 1, 4, "rint", TOKEN_PRINT);
118 case 'r': return check_keyword(lexer, 1, 5, "eturn", TOKEN_RETURN);
119 case 's': return check_keyword(lexer, 1, 4, "uper", TOKEN_SUPER);
120 case 't':
121 if (lexer->current - lexer->start > 1) {
122 switch (lexer->start[1]) {
123 case 'h': return check_keyword(lexer, 2, 2, "is", TOKEN_THIS);
124 case 'r': return check_keyword(lexer, 2, 2, "ue", TOKEN_TRUE);
125 }
126 }
127 break;
128 case 'v': return check_keyword(lexer, 1, 2, "ar", TOKEN_VAR);
129 case 'w': return check_keyword(lexer, 1, 4, "hile", TOKEN_WHILE);
130 }
131 return TOKEN_IDENTIFIER;
132}

134static bool is_digit(char c) {
135 return c >= '0' && c <= '9';
136}
137
138static bool is_alpha(char c) {
139 return (c >= 'a' && c <= 'z') ||
140 (c >= 'A' && c <= 'Z') ||
141 c == '_';
142}

144// Scans an identifier or keyword.
145static Token identifier(Lexer* lexer) {
146 while (is_alpha(peek(lexer)) || is_digit(peek(lexer))) advance(lexer);
147 return make_token(lexer, identifier_type(lexer));
148}
149
150// Scans a number literal (integer or float).
151static Token number(Lexer* lexer) {
152 while (is_digit(peek(lexer))) advance(lexer);
153
154 // Look for a fractional part.
155 if (peek(lexer) == '.' && is_digit(peek_next(lexer))) {
156 // Consume the ".".
157 advance(lexer);
158 while (is_digit(peek(lexer))) advance(lexer);
159 return make_token(lexer, TOKEN_FLOAT_LITERAL);
160 }
161
162 return make_token(lexer, TOKEN_INT_LITERAL);
163}
164
165// Scans a string literal.
166static Token string(Lexer* lexer) {
167 while (peek(lexer) != '"' && !is_at_end(lexer)) {
168 if (peek(lexer) == '\n') lexer->line++;
169 advance(lexer);
170 }
171
172 if (is_at_end(lexer)) return error_token(lexer, "Unterminated string.");
173
174 // The closing quote.
175 advance(lexer);
176 return make_token(lexer, TOKEN_STRING);
177}

188Token lexer_scan_token(Lexer* lexer) {
189 skip_whitespace(lexer);
190 lexer->start = lexer->current;
191
192 if (is_at_end(lexer)) return make_token(lexer, TOKEN_EOF);
193
194 char c = advance(lexer);
195
196 if (is_alpha(c)) return identifier(lexer);
197 if (is_digit(c)) return number(lexer);
198
199 switch (c) {
200 case '(': return make_token(lexer, TOKEN_LPAREN);
201 case ')': return make_token(lexer, TOKEN_RPAREN);
202 case '{': return make_token(lexer, TOKEN_LBRACE);
203 case '}': return make_token(lexer, TOKEN_RBRACE);
204 case '[': return make_token(lexer, TOKEN_LBRACKET);
205 case ']': return make_token(lexer, TOKEN_RBRACKET);
206 case ';': return make_token(lexer, TOKEN_SEMICOLON);
207 case ',': return make_token(lexer, TOKEN_COMMA);
208 case '.': return make_token(lexer, TOKEN_DOT);
209 case '-': return make_token(lexer, TOKEN_MINUS);
210 case '+': return make_token(lexer, TOKEN_PLUS);
211 case '/': return make_token(lexer, TOKEN_SLASH);
212 case '*': return make_token(lexer, TOKEN_STAR);
213 case '!': return make_token(lexer, match(lexer, '=') ? TOKEN_BANG_EQUAL : TOKEN_BANG);
214 case '=': return make_token(lexer, match(lexer, '=') ? TOKEN_EQUAL_EQUAL : TOKEN_EQUAL);
215 case '<': return make_token(lexer, match(lexer, '=') ? TOKEN_LESS_EQUAL : TOKEN_LESS);
216 case '>': return make_token(lexer, match(lexer, '=') ? TOKEN_GREATER_EQUAL : TOKEN_GREATER);
217 case '"': return string(lexer);
218 }
219
220 return error_token(lexer, "Unexpected character.");
221}
