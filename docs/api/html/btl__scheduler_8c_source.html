102void btl_scheduler_add_dependency(BTL_Scheduler *scheduler, uint32_t from_id,
103 uint32_t to_id) {
104 if (!scheduler || from_id >= scheduler->graph.num_nodes ||
105 to_id >= scheduler->graph.num_nodes) return;
106
107 BTL_InstructionNode *from = scheduler->graph.nodes[from_id];
108 BTL_InstructionNode *to = scheduler->graph.nodes[to_id];
109
110 // Add to successors of 'from'
111 if (from->num_successors >= from->successors_capacity) {
112 size_t new_capacity = from->successors_capacity * 2;
113 BTL_InstructionNode **new_successors = realloc(from->successors,
114 new_capacity * sizeof(BTL_InstructionNode*));
115 if (!new_successors) return;
116 from->successors = new_successors;
117 from->successors_capacity = new_capacity;
118 }
119 from->successors[from->num_successors++] = to;
120
121 // Add to predecessors of 'to'
122 if (to->num_predecessors >= to->predecessors_capacity) {
123 size_t new_capacity = to->predecessors_capacity * 2;

128 to->predecessors_capacity = new_capacity;

133// Compute earliest start times (forward pass)
134static void compute_earliest_start(BTL_Scheduler *scheduler) {
135 for (size_t i = 0; i < scheduler->graph.num_nodes; i++) {
136 BTL_InstructionNode *node = scheduler->graph.nodes[i];
137 node->earliest_start = 0;
138
139 // Find maximum of predecessor finish times
140 for (size_t j = 0; j < node->num_predecessors; j++) {
141 BTL_InstructionNode *pred = node->predecessors[j];
142 uint32_t pred_finish = pred->earliest_start + pred->latency;
143 if (pred_finish > node->earliest_start) {
144 node->earliest_start = pred_finish;
145 }
146 }
147 }
148}

150// Compute latest start times (backward pass)
151static void compute_latest_start(BTL_Scheduler *scheduler, uint32_t makespan) {
152 // Initialize all to makespan
153 for (size_t i = 0; i < scheduler->graph.num_nodes; i++) {
154 scheduler->graph.nodes[i]->latest_start = makespan;
155 }
156
157 // Backward pass
158 for (size_t i = scheduler->graph.num_nodes; i > 0; i--) {
159 BTL_InstructionNode *node = scheduler->graph.nodes[i - 1];
160
161 // Find minimum of successor start times minus latency
162 for (size_t j = 0; j < node->num_successors; j++) {
163 BTL_InstructionNode *succ = node->successors[j];
164 uint32_t required_start = succ->latest_start - node->latency;
165 if (required_start < node->latest_start) {
166 node->latest_start = required_start;
167 }
168 }
169 }
170}

172bool btl_scheduler_build_graph(BTL_Scheduler *scheduler) {
173 if (!scheduler) return false;
174
175 compute_earliest_start(scheduler);
176
177 // Find makespan (maximum earliest finish time)
178 uint32_t makespan = 0;
179 for (size_t i = 0; i < scheduler->graph.num_nodes; i++) {
180 BTL_InstructionNode *node = scheduler->graph.nodes[i];
181 uint32_t finish = node->earliest_start + node->latency;
182 if (finish > makespan) makespan = finish;
183 }
184
185 compute_latest_start(scheduler, makespan);
186 scheduler->critical_path_length = makespan;
187
188 return true;
189}

191// List scheduling algorithm
192bool btl_scheduler_schedule(BTL_Scheduler *scheduler) {
193 if (!scheduler) return false;
194
195 // Build graph first
196 if (!btl_scheduler_build_graph(scheduler)) return false;
197
198 // Ready list: instructions with no unscheduled predecessors
199 BTL_InstructionNode **ready = calloc(scheduler->graph.num_nodes,
200 sizeof(BTL_InstructionNode*));
201 if (!ready) return false;
202
203 size_t ready_count = 0;
204 uint32_t current_time = 0;
205 size_t scheduled_count = 0;
206
207 // Initialize ready list with instructions that have no predecessors
208 for (size_t i = 0; i < scheduler->graph.num_nodes; i++) {
209 BTL_InstructionNode *node = scheduler->graph.nodes[i];
210 if (node->num_predecessors == 0) {
211 ready[ready_count++] = node;
212 }
213 }
214
215 // Schedule instructions
216 while (scheduled_count < scheduler->graph.num_nodes) {
217 if (ready_count == 0) {
218 // Advance time to next instruction completion
219 current_time++;
220
221 // Check for newly ready instructions
222 for (size_t i = 0; i < scheduler->graph.num_nodes; i++) {
223 BTL_InstructionNode *node = scheduler->graph.nodes[i];
224 if (node->scheduled) continue;
225
226 bool all_preds_done = true;
227 for (size_t j = 0; j < node->num_predecessors; j++) {
228 BTL_InstructionNode *pred = node->predecessors[j];
229 if (!pred->scheduled ||
230 pred->schedule_time + pred->latency > current_time) {
231 all_preds_done = false;
232 break;
233 }
234 }
235
236 if (all_preds_done) {
237 ready[ready_count++] = node;
238 }
239 }
240
241 continue;
242 }
243
244 // Select instruction with earliest start time (priority)
245 size_t best_idx = 0;
246 for (size_t i = 1; i < ready_count; i++) {
247 if (ready[i]->earliest_start < ready[best_idx]->earliest_start) {
248 best_idx = i;
249 }
250 }
251
252 BTL_InstructionNode *node = ready[best_idx];
253
254 // Expand schedule capacity if needed
255 if (scheduler->schedule_count >= scheduler->schedule_capacity) {
256 size_t new_capacity = scheduler->schedule_capacity * 2;
257 uint32_t *new_schedule = realloc(scheduler->schedule,
258 new_capacity * sizeof(uint32_t));
259 if (!new_schedule) {
260 free(ready);
261 return false;
262 }
263 scheduler->schedule = new_schedule;
264 scheduler->schedule_capacity = new_capacity;
265 }
266
267 // Schedule the instruction
268 node->scheduled = true;
269 node->schedule_time = current_time;
270 scheduler->schedule[scheduler->schedule_count++] = node->id;
271 scheduled_count++;
272
273 // Remove from ready list
274 ready[best_idx] = ready[--ready_count];
275 }
276
277 free(ready);
278 return true;
279}
