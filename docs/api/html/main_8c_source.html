18// This function encapsulates the entire compiler and VM pipeline.
19static InterpretResult compile_and_run(const char* source) {
20 // 1. Lexer
21 Lexer lexer;
22 lexer_init(&lexer, source);
23
24 // 2. Parser
25 Parser parser;
26 parser_init(&parser, &lexer);
27 AstNode* program = parser_parse(&parser);
28 if (program == nullptr || parser.had_error) {
29 fprintf(stderr, "Compilation failed: Syntax Error.\n");
30 if (program) ast_free_node(program);
31 return INTERPRET_COMPILE_ERROR;
32 }
33
34 // 3. Semantic Analyzer
35 Analyzer analyzer;
36 analyzer_init(&analyzer);
37 bool is_semantically_valid = analyzer_analyze(&analyzer, program);
38 if (!is_semantically_valid) {
39 fprintf(stderr, "Compilation failed: Semantic Error.\n");
40 analyzer_free(&analyzer);
41 ast_free_node(program);
42 return INTERPRET_COMPILE_ERROR;
43 }
44
45 // 4. Code Generator
46 Chunk chunk;
47 chunk_init(&chunk);
48 CodeGenerator codegen;
49 codegen_init(&codegen, &chunk);
50 bool compilation_success = codegen_generate(&codegen, program);
51 if (!compilation_success) {
52 fprintf(stderr, "Compilation failed: Code Generation Error.\n");
53 codegen_free(&codegen);
54 chunk_free(&chunk);
55 analyzer_free(&analyzer);
56 ast_free_node(program);
57 return INTERPRET_COMPILE_ERROR;
58 }
59
60 // 5. Virtual Machine
61 VM vm;
62 vm_init(&vm);
63 InterpretResult result = vm_interpret(&vm, &chunk);
64
65 // 6. Cleanup
66 vm_free(&vm);
67 codegen_free(&codegen);
68 chunk_free(&chunk);
69 analyzer_free(&analyzer);
70 ast_free_node(program);
71
72 return result;
73}
