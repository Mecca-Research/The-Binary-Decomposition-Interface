59float ham_compute_numa_affinity(const HamRegion* region, uint32_t node_id) {
60 if (!region) return 0.0f;
61
62 // Compute affinity based on:
63 // 1. Access frequency
64 // 2. Memory tier
65 // 3. Distance to node
66
67 float affinity = 0.0f;
68
69 // Higher access count = higher affinity
70 affinity += (float)region->stats.access_count / 10000.0f;
71
72 // Critical tier = higher affinity
73 switch (region->tier) {
74 case HAM_CRITICAL:
75 affinity += 1.0f;
76 break;
77 case HAM_ACTIVE:
78 affinity += 0.5f;
79 break;
80 case HAM_DORMANT:
81 affinity += 0.2f;
82 break;
83 case HAM_ARCHIVE:
84 affinity += 0.1f;
85 break;
86 }
87
88 // Normalize
89 if (affinity > 1.0f) affinity = 1.0f;
90
91 return affinity;

112bool should_migrate(const HamRegion* region, uint32_t current_node, uint32_t target_node) {
113 if (!region || current_node == target_node) return false;

115 // Migrate if:

122 // Require 20% improvement to justify migration cost
