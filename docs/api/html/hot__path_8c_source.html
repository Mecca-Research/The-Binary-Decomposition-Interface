171static int compare_hotness(const void* a, const void* b) {
172 const HotPathInfo* info_a = (const HotPathInfo*)a;
173 const HotPathInfo* info_b = (const HotPathInfo*)b;
174
175 if (info_a->hotness_score > info_b->hotness_score) return -1;

180size_t hot_path_detector_get_hot_paths(
181 const HotPathDetector* detector,
182 HotPathInfo* paths,
183 size_t max_paths
184) {
185 if (!detector || !paths || max_paths == 0) return 0;
186
187 size_t count = 0;
188 for (size_t i = 0; i < detector->entry_count && count < max_paths; i++) {
189 const HotPathEntry* entry = &detector->entries[i];
190 if (entry->is_hot) {
191 paths[count].function_id = entry->function_id;
192 paths[count].basic_block_id = entry->basic_block_id;
193 paths[count].execution_count = entry->execution_count;
194 paths[count].hotness_score = (double)entry->execution_count /
195 (entry->total_time_ns + 1);
196 count++;
197 }
198 }
199
200 // Sort by hotness score
201 qsort(paths, count, sizeof(HotPathInfo), compare_hotness);
202
203 return count;
204}
