12float ham_compute_entropy(const void* data, size_t size) {
13 if (!data || size == 0) return 0.0f;
14
15 // Count byte frequencies
16 uint32_t freq[256] = {0};
17 const uint8_t* bytes = (const uint8_t*)data;
18
19 for (size_t i = 0; i < size; i++) {
20 freq[bytes[i]]++;
21 }
22
23 // Compute Shannon entropy
24 float entropy = 0.0f;
25
26 for (int i = 0; i < 256; i++) {
27 if (freq[i] > 0) {
28 float p = (float)freq[i] / (float)size;
29 entropy -= p * log2f(p);
30 }
31 }
32
33 return entropy;
34}

89float access_pattern_compute_entropy(const AccessPattern* pattern) {
90 if (!pattern || pattern->history_size == 0) return 0.0f;
91
92 // Compute inter-access time deltas
93 if (pattern->history_size < 2) return 0.0f;
94
95 uint64_t* deltas = malloc(sizeof(uint64_t) * (pattern->history_size - 1));
96 if (!deltas) return 0.0f;
97
98 for (size_t i = 1; i < pattern->history_size; i++) {
99 deltas[i - 1] = pattern->access_history[i] - pattern->access_history[i - 1];
100 }
101
102 // Quantize deltas into buckets
103 uint32_t buckets[16] = {0};
104
105 for (size_t i = 0; i < pattern->history_size - 1; i++) {
106 uint64_t delta = deltas[i];
107 int bucket = 0;
108
109 if (delta < 10) bucket = 0;
110 else if (delta < 100) bucket = 1;
111 else if (delta < 1000) bucket = 2;
112 else bucket = 3;
113
114 buckets[bucket]++;
115 }
116
117 // Compute entropy
118 float entropy = 0.0f;
119 size_t total = pattern->history_size - 1;
120
121 for (int i = 0; i < 16; i++) {
122 if (buckets[i] > 0) {
123 float p = (float)buckets[i] / (float)total;
124 entropy -= p * log2f(p);
125 }
126 }
127
128 free(deltas);
129 return entropy;
130}
