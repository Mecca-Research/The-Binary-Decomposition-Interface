6MarkSweepGC* mark_sweep_create(size_t heap_size) {
7 MarkSweepGC* gc = (MarkSweepGC*)calloc(1, sizeof(MarkSweepGC));
8 if (!gc) return NULL;
9
10 gc->heap_start = (GCObject*)malloc(heap_size);

12 free(gc);

42GCObject* mark_sweep_allocate(MarkSweepGC* gc, size_t size, uint32_t type_id) {
43 if (!gc || size == 0) return NULL;
44
45 // Align size to 8 bytes
46 size = (size + 7) & ~7;
47 size_t total_size = size + sizeof(GCObjectHeader);
48
49 // Find suitable free block
50 GCObject* prev = NULL;
51 GCObject* current = gc->free_list;
52
53 while (current) {
54 if (current->header.size >= size) {
55 // Found suitable block
56 GCObject* allocated = current;
57
58 // Split block if large enough
59 if (current->header.size >= total_size + sizeof(GCObjectHeader) + 64) {
60 GCObject* remainder = (GCObject*)((uint8_t*)current + total_size);
61 remainder->header.size = current->header.size - total_size;
62 remainder->header.marked = false;
63 remainder->header.pinned = false;
64 remainder->header.next = current->header.next;
65
66 if (prev) {
67 prev->header.next = remainder;
68 } else {
69 gc->free_list = remainder;
70 }
71 } else {
72 // Use entire block
73 if (prev) {
74 prev->header.next = current->header.next;
75 } else {
76 gc->free_list = current->header.next;
77 }
78 gc->free_count--;
79 }
80
81 allocated->header.size = size;
82 allocated->header.type_id = type_id;
83 allocated->header.marked = false;
84 allocated->header.pinned = false;
85 allocated->header.next = NULL;
86
87 gc->heap_used += total_size;
88 gc->objects_allocated++;
89 gc->bytes_allocated += size;
90
91 return allocated;
92 }
93
94 prev = current;
95 current = current->header.next;
96 }
97
98 return NULL;
99}

101void mark_sweep_free(MarkSweepGC* gc, GCObject* object) {
102 if (!gc || !object) return;
103
104 size_t total_size = object->header.size + sizeof(GCObjectHeader);
105
106 // Add to free list
107 object->header.next = gc->free_list;
108 gc->free_list = object;
109 gc->free_count++;
110
111 gc->heap_used -= total_size;
112 gc->objects_freed++;

134static void sweep_phase(MarkSweepGC* gc) {
135 if (!gc) return;
136
137 GCObject* current = gc->heap_start;
138
139 while ((uint8_t*)current < (uint8_t*)gc->heap_end) {
140 if (!current->header.marked && !current->header.pinned) {
141 // Object is garbage, free it
142 mark_sweep_free(gc, current);
143 } else {
144 // Clear mark for next collection
145 current->header.marked = false;
146 }
147
148 // Move to next object
149 size_t total_size = current->header.size + sizeof(GCObjectHeader);
150 current = (GCObject*)((uint8_t*)current + total_size);
151 }
152}

154bool mark_sweep_collect(MarkSweepGC* gc, GCRootSet* roots) {
155 if (!gc || !roots) return false;
156
157 // BUG FIX (P1): Clear free list before rebuilding it in sweep_phase
158 // Without this, sweep_phase appends to the existing list, causing
159 // duplicate entries and corrupted heap_used accounting on repeated GCs
160 gc->free_list = NULL;
161 gc->free_count = 0;
162
163 mark_phase(gc, roots);
164 sweep_phase(gc);
165
166 gc->collections++;
167
168 // Update threshold
169 gc->gc_threshold = (size_t)(gc->heap_used * gc->gc_growth_factor);
170
171 return true;
172}

202bool gc_root_set_add(GCRootSet* roots, GCObject* object) {
203 if (!roots || !object) return false;
204
205 // Check if already in root set
206 for (size_t i = 0; i < roots->root_count; i++) {
207 if (roots->roots[i] == object) return true;
208 }
209
210 // Expand if needed
211 if (roots->root_count >= roots->root_capacity) {
212 size_t new_capacity = roots->root_capacity * 2;
213 GCObject** new_roots = (GCObject**)realloc(roots->roots,
214 new_capacity * sizeof(GCObject*));
215 if (!new_roots) return false;
216
217 roots->roots = new_roots;
218 roots->root_capacity = new_capacity;
219 }
220
221 roots->roots[roots->root_count++] = object;
222 return true;
223}

225bool gc_root_set_remove(GCRootSet* roots, GCObject* object) {
226 if (!roots || !object) return false;
227
228 for (size_t i = 0; i < roots->root_count; i++) {
229 if (roots->roots[i] == object) {
230 // Shift remaining elements
231 memmove(&roots->roots[i], &roots->roots[i + 1],
232 (roots->root_count - i - 1) * sizeof(GCObject*));
233 roots->root_count--;
234 return true;
235 }
236 }

238 return false;
