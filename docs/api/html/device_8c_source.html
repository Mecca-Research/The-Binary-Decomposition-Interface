15static void cpu_kernel_const(void** inputs, void* outputs[1]) {
16 *(float*)outputs[0] = *(float*)inputs[0] + *(float*)inputs[1];
17 // The "constant" value is often embedded in the kernel or a metadata region.
18 // For M0, we assume it's pre-loaded into the output region by the scheduler.
19 // This function is effectively a no-op for this simple model.
20}

22static void cpu_kernel_add_f32(void** inputs, void* outputs[1]) {
23 float* in1 = (float*)inputs[0];
24 float* in2 = (float*)inputs[1];
25 float* out = (float*)outputs[0];
26 *out = *in1 + *in2;
27}

29static void cpu_kernel_mul_f32(void** inputs, void* outputs[1]) {
30 float* in1 = (float*)inputs[0];
31 float* in2 = (float*)inputs[1];
32 float* out = (float*)outputs[0];
33 *out = *in1 * *in2;

36static void cpu_kernel_relu_f32(void** inputs, void* outputs[1]) {
37 float* in = (float*)inputs[0];
38 float* out = (float*)outputs[0];
39 *out = *in > 0 ? *in : 0;
40}

42// --- Learning Kernels ---
43// A simplified gradient calculation for a single neuron: grad_in = grad_out * weight
44static void cpu_kernel_grad_f32(void** inputs, void* outputs[1]) {
45 float* grad_out = (float*)inputs[0]; // Gradient from the next layer
46 float* weight = (float*)inputs[1]; // Weight of the connection
47 float* grad_in = (float*)outputs[0]; // Gradient for the previous layer
48 *grad_in = *grad_out * *weight;
49 // Note: A full backprop implementation is much more complex, involving derivatives
50 // of activation functions, but this demonstrates the data flow.
51}

53// A simplified parameter update: param = param - learning_rate * grad
54static void cpu_kernel_update_f32(void** inputs, void* outputs[1]) {
55 float* param = (float*)inputs[0]; // The parameter to update (e.g., a weight)
56 float* grad = (float*)inputs[1]; // The calculated gradient for this parameter
57 float* lr = (float*)inputs[2]; // The learning rate
58
59 *param -= *lr * *grad;
60 // This kernel modifies its input region directly (side-effect).
61 // The output is not used.
62}

66static int cpu_lower(const GraphNode* node, void* out_kernel) {
67 // This function maps an OpCode to a specific kernel function pointer.
68 void (**kernel_ptr)(void**, void**) = (void (**)(void**, void**))out_kernel;
69
70 switch(node->op) {
71 case OP_CONST:
72 *kernel_ptr = cpu_kernel_const;
73 break;
74 case OP_ADD:
75 // In a real system, you'd check node->out_type to select the correct kernel.
76 *kernel_ptr = cpu_kernel_add_f32;
77 break;
78 case OP_MUL:
79 *kernel_ptr = cpu_kernel_mul_f32;
80 break;
81 case OP_RELU:
82 *kernel_ptr = cpu_kernel_relu_f32;
83 break;
84 // OP_MATMUL would require a more complex kernel.
85 case OP_GRAD: *kernel_ptr = cpu_kernel_grad_f32; break;
86 case OP_UPDATE: *kernel_ptr = cpu_kernel_update_f32; break;
87 default:
88 fprintf(stderr, "CPU_DEVICE Error: Cannot lower OpCode %d\n", node->op);
89 return -1; // Failure
90 }
91 return 0; // Success
92}

94static int cpu_enqueue(const void* kernel, const HamRegion** regions, size_t num_regions) {
95 // For a simple, synchronous CPU, "enqueue" means "execute immediately."
96 void (*kernel_func)(void**, void**) = (void (*)(void**, void**))kernel;
97
98 // A real scheduler would map node inputs/outputs to these region pointers.
99 // This is a placeholder for that complex logic.
100 void* inputs[8] = {0};
101 void* outputs[1] = {0};
102
103 // Example: Assume region 0 is input 1, region 1 is input 2, region 2 is output.
104 if (num_regions > 0) inputs[0] = regions[0]->base;
105 if (num_regions > 1) inputs[1] = regions[1]->base;
106 if (num_regions > 2) outputs[0] = regions[2]->base;
107 // For many ops, the last region is the output
108 if (num_regions > 0) outputs[0] = regions[num_regions-1]->base;
109 printf("CPU_DEVICE: Executing kernel...\n");
110 kernel_func(inputs, outputs);
111
112 return 0; // Success
113}

121static int gpu_enqueue(const void* kernel, const HamRegion** regions, size_t num_regions) {
122 // This function is the bridge to the gpu_backend.c API.
123 GpuKernel gpu_kernel = *(GpuKernel*)kernel;
124
125 // A real implementation would involve:
126 // 1. Allocating GPU memory for each region (gpu_alloc).
127 // 2. Copying input data from host (HAM region) to device (gpu_memcpy_h2d).
128 // 3. Setting up kernel launch arguments (an array of device pointers).
129 // 4. Launching the kernel (gpu_launch_kernel).
130 // 5. Copying results back from device to host (gpu_memcpy_d2h).
131 // 6. Freeing GPU memory (gpu_free).
132
133 // For our M3 test, we will just simulate the launch.
134 return gpu_launch_kernel(gpu_kernel, nullptr);
135}

151// The 'enqueue' function for the FPGA is the bitstream loading and execution step.
152static int fpga_enqueue(const void* kernel, const HamRegion** regions, size_t num_regions) {
153 const GraphNode* subgraph_start_node = *(const GraphNode**)kernel;
154
155 // A real implementation would traverse the graph from this start node
156 // to find the end of the subgraph. For M5, we'll assume it's just this one node.
157 FpgaBitstream* bitstream = fpga_synthesize_subgraph(nullptr, subgraph_start_node->id, subgraph_start_node->id);
158 if (!bitstream) {
159 fprintf(stderr, "FPGA_DEVICE Error: Synthesis failed.\n");
160 return -1;
161 }
162
163 int result = fpga_load_bitstream(bitstream);
164
165 // After loading, a real implementation would copy data from HAM regions to the FPGA,
166 // trigger the execution, and copy results back.
167
168 fpga_free_bitstream(bitstream);
169 return result;
170}
