12char* generate_verilog_add(const GraphNode* node) {
13 if (!node) return NULL;
14
15 const char* template =
16 "module add_%llu (\n"
17 " input wire clk,\n"
18 " input wire rst,\n"
19 " input wire [31:0] a,\n"
20 " input wire [31:0] b,\n"
21 " output reg [31:0] c\n"
22 ");\n"
23 " always @(posedge clk) begin\n"
24 " if (rst) c <= 32'b0;\n"
25 " else c <= a + b;\n"
26 " end\n"
27 "endmodule\n";
28
29 size_t len = strlen(template) + 64;
30 char* code = malloc(len);

88int fpga_synthesize_subgraph(const BdiGraph* graph, const Subgraph* subgraph, FpgaKernel** out_kernel) {
89 if (!graph || !subgraph || !out_kernel) return -1;
90
91 FpgaKernel* kernel = malloc(sizeof(FpgaKernel));
92 if (!kernel) return -1;
93
94 // Allocate buffer for combined Verilog code
95 size_t total_len = 4096;
96 kernel->verilog_code = malloc(total_len);
97 if (!kernel->verilog_code) {
98 free(kernel);
99 return -1;
100 }
101 kernel->verilog_code[0] = '\0'; // Initialize buffer to empty string
102
103 kernel->code_len = 0;
104 kernel->latency_cycles = 0;
105 kernel->resource_usage = 0;
106
107 // Generate Verilog for each node in subgraph
108 for (size_t i = 0; i < subgraph->count; i++) {
109 NodeId node_id = subgraph->nodes[i];
110 if (node_id >= graph->node_count) continue;
111
112 const GraphNode* node = &graph->nodes[node_id];
113 char* node_verilog = NULL;
114
115 switch (node->op) {
116 case OP_ADD:
117 node_verilog = generate_verilog_add(node);
118 kernel->latency_cycles += 1;
119 kernel->resource_usage += 100;
120 break;
121 case OP_MUL:
122 node_verilog = generate_verilog_mul(node);
123 kernel->latency_cycles += 3;
124 kernel->resource_usage += 300;
125 break;
126 case OP_RELU:
127 node_verilog = generate_verilog_relu(node);
128 kernel->latency_cycles += 1;
129 kernel->resource_usage += 50;
130 break;
131 default:
132 continue;
133 }
134
135 if (node_verilog) {
136 size_t node_len = strlen(node_verilog);
137 if (kernel->code_len + node_len + 1 > total_len) {
138 total_len *= 2;
139 char* new_code = realloc(kernel->verilog_code, total_len);
140 if (!new_code) {
141 free(kernel->verilog_code);
142 free(kernel);
143 return -1;
144 }
145 kernel->verilog_code = new_code;
146 }
147
148 strcat(kernel->verilog_code, node_verilog);
149 kernel->code_len += node_len;
150 free(node_verilog);
151 }
152 }
153
154 *out_kernel = kernel;
155 return 0;
156}

184 kernel->resource_usage = 50;

192 kernel->code_len = strlen(kernel->verilog_code);
193 }
194
