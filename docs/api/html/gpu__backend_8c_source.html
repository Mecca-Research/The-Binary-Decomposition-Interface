30void* gpu_alloc(size_t size_bytes) {
31 if (!gpu_initialized || gpu_mem_allocated + size_bytes > GPU_MEMORY_CAPACITY) {
32 return nullptr;
33 }
34 // In a real implementation, this would return a device pointer.
35 // We'll return a host pointer from malloc to simulate it.
36 void* ptr = malloc(size_bytes);
37 if (ptr) {
38 gpu_mem_allocated += size_bytes;
39 printf("GPU_BACKEND: Allocated %zu bytes on device.\n", size_bytes);
40 }
41 return ptr;
42}
