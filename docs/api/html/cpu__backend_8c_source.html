41void cpu_kernel_matmul(const void* inputs[], void* output, size_t count) {
42 // Simplified matrix multiplication (assumes square matrices)
43 const float* a = (const float*)inputs[0];
44 const float* b = (const float*)inputs[1];
45 float* c = (float*)output;
46
47 size_t n = (size_t)sqrt((double)count);
48
49 for (size_t i = 0; i < n; i++) {
50 for (size_t j = 0; j < n; j++) {
51 float sum = 0.0f;
52 for (size_t k = 0; k < n; k++) {
53 sum += a[i * n + k] * b[k * n + j];
54 }
55 c[i * n + j] = sum;
56 }
57 }
58}

62int cpu_lower(const GraphNode* node, void** out_kernel) {
63 if (!node || !out_kernel) return -1;
64
65 CpuKernel* kernel = malloc(sizeof(CpuKernel));
66 if (!kernel) return -1;
67
68 kernel->opcode = node->op;
69 kernel->type = node->out_type;
70 kernel->input_count = node->input_count;
71 kernel->metadata = NULL;
72
73 // Map opcode to kernel function
74 switch (node->op) {
75 case OP_ADD:
76 kernel->execute = cpu_kernel_add;
77 break;
78 case OP_MUL:
79 kernel->execute = cpu_kernel_mul;
80 break;
81 case OP_RELU:
82 kernel->execute = cpu_kernel_relu;
83 break;
84 case OP_MATMUL:
85 kernel->execute = cpu_kernel_matmul;
86 break;
87 default:
88 free(kernel);
89 return -1;
90 }
91
92 *out_kernel = kernel;
93 return 0;
94}

96int cpu_enqueue(const void* kernel, const HamRegion** regions, size_t num_regions) {
97 if (!kernel || !regions) return -1;
98
99 const CpuKernel* cpu_kernel = (const CpuKernel*)kernel;
100
101 // Prepare input pointers
102 const void* inputs[8] = {0};
103 for (size_t i = 0; i < cpu_kernel->input_count && i < num_regions - 1; i++) {
104 inputs[i] = regions[i]->base;
105 }
106
107 // Output is the last region
108 void* output = (void*)regions[num_regions - 1]->base;
109
110 // Execute kernel
111 size_t count = regions[num_regions - 1]->capacity_bytes / sizeof(float);
112 cpu_kernel->execute(inputs, output, count);
113
114 return 0;
115}
