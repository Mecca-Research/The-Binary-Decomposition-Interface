41// Main dispatcher function to generate code for any AST node.
42static void generate_node(CodeGenerator* codegen, AstNode* node) {
43 if (!node) return;
44
45 switch (node->kind) {
46 case AST_NODE_LITERAL: {
47 // Our VM currently uses doubles for all values.
48 emit_constant(codegen, (double)node->as.literal.value.i64, node->line);
49 break;
50 }
51 case AST_NODE_BINARY_OP: {
52 generate_node(codegen, node->as.binary_op.left);
53 generate_node(codegen, node->as.binary_op.right);
54
55 const char* op = node->as.binary_op.op;
56 if (strcmp(op, "+") == 0) emit_byte(codegen, OP_ADD, node->line);
57 else if (strcmp(op, "-") == 0) emit_byte(codegen, OP_SUBTRACT, node->line);
58 else if (strcmp(op, "*") == 0) emit_byte(codegen, OP_MULTIPLY, node->line);
59 else if (strcmp(op, "/") == 0) emit_byte(codegen, OP_DIVIDE, node->line);
60 // Add cases for comparison operators later.
61 break;
62 }
63 case AST_NODE_UNARY_OP: {
64 if (node->as.binary_op.op && strcmp(node->as.binary_op.op, "-") == 0) {
65 generate_node(codegen, node->as.binary_op.right);
66 emit_byte(codegen, OP_NEGATE, node->line);
67 }
68 // Add other unary ops like '!' later.
69 break;
70 }
71 case AST_NODE_PROGRAM:
72 case AST_NODE_BLOCK: {
73 for (size_t i = 0; i < node->as.block->len; ++i) {
74 generate_node(codegen, node->as.block->data[i]);
75 }
76 break;
77 }
78 // Add cases for statements and declarations as they are added to the language.
79 default:
80 fprintf(stderr, "Codegen Error: Unknown AST node kind %d\n", node->kind);
81 codegen->had_error = true;
82 break;
83 }
84}
