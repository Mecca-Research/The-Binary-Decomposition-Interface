23// Handles a binary operation node.
24static void analyze_binary_op(Analyzer* analyzer, AstNode* node) {
25 analyze_node(analyzer, node->as.binary_op.left);
26 analyze_node(analyzer, node->as.binary_op.right);
27
28 // Basic type check: for now, we only support operations on integers.
29 BciType* left_type = node->as.binary_op.left->resolved_type;
30 BciType* right_type = node->as.binary_op.right->resolved_type;
31
32 if (left_type && right_type) {
33 if (left_type->kind != BCI_TYPE_I64 || right_type->kind != BCI_TYPE_I64) {
34 error(analyzer, node->line, "Operands must be integers.");
35 }
36 }
37
38 // The result of an arithmetic operation is an integer.
39 // A proper implementation would have a type promotion system.
40 node->resolved_type = malloc(sizeof(BciType));
41 node->resolved_type->kind = BCI_TYPE_I64;
42 node->resolved_type->base = nullptr;
43}

53// Main dispatcher for traversing any AST node.
54static void analyze_node(Analyzer* analyzer, AstNode* node) {
55 if (!node) return;
56
57 switch (node->kind) {
58 case AST_NODE_LITERAL:
59 analyze_literal(analyzer, node);
60 break;
61 case AST_NODE_BINARY_OP:
62 analyze_binary_op(analyzer, node);
63 break;
64 case AST_NODE_BLOCK:
65 case AST_NODE_PROGRAM:
66 analyze_block(analyzer, node->as.block);
67 break;
68 // Other cases for statements, declarations, etc. will go here.
69 default:
70 // This case should not be reached in a complete analyzer.
71 break;
72 }
73}

75// Traverses a block of statements within a new scope.
76static void analyze_block(Analyzer* analyzer, AstBlock* block) {
77 symbol_table_begin_scope(analyzer->symbol_table);
78 for (size_t i = 0; i < block->len; ++i) {
79 analyze_node(analyzer, block->data[i]);
80 }
81 symbol_table_end_scope(analyzer->symbol_table);
82}
