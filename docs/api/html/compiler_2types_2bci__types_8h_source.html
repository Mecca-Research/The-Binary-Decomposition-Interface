72#define bci_vec_init(v) do { (v)->data = nullptr; (v)->len = 0; (v)->capacity = 0; } while (0)
73#define bci_vec_free(v) free((v)->data)
74#define bci_vec_push(v, val) do { \
75 if ((v)->len >= (v)->capacity) { \
76 (v)->capacity = ((v)->capacity == 0) ? 8 : (v)->capacity * 2; \
77 (v)->data = realloc((v)->data, (v)->capacity * sizeof(*(v)->data)); \
78 } \
79 (v)->data[(v)->len++] = (val); \
80} while (0)
