11// Simple hash function for data blocks (e.g., djb2)
12static uint64_t hash_data(const void* data, size_t size) {
13 uint64_t hash = 5381;
14 const unsigned char* str = (const unsigned char*)data;
15 for (size_t i = 0; i < size; i++) {
16 hash = ((hash << 5) + hash) + str[i]; // hash * 33 + c
17 }
18 return hash;
19}

37Motif* motif_dict_intern(MotifDictionary* dict, void* data, size_t size) {
38 if (!dict || !data || size == 0) return nullptr;
39
40 uint64_t hash = hash_data(data, size);
41
42 // 1. Check if a motif with this hash and data already exists.
43 for (size_t i = 0; i < dict->count; i++) {
44 if (dict->motifs[i].hash == hash &&
45 dict->motifs[i].size == size &&
46 memcmp(dict->motifs[i].data, data, size) == 0)
47 {
48 dict->motifs[i].ref_count++;
49 return &dict->motifs[i];
50 }
51 }
52
53 // 2. If not found, create a new motif.
54 if (dict->count >= dict->capacity) {
55 size_t new_capacity = dict->capacity < 8 ? 8 : dict->capacity * 2;
56 Motif* new_motifs = (Motif*)realloc(dict->motifs, new_capacity * sizeof(Motif));
57 if (!new_motifs) return nullptr;
58 dict->motifs = new_motifs;
59 dict->capacity = new_capacity;
60 }
61
62 void* motif_data = malloc(size);
63 if (!motif_data) return nullptr;
64 memcpy(motif_data, data, size);
65
66 dict->motifs[dict->count] = (Motif){
67 .hash = hash,
68 .data = motif_data,
69 .size = size,
70 .ref_count = 1
71 };
72
73 return &dict->motifs[dict->count++];
74}
