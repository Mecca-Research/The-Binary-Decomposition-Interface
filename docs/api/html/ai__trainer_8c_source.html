58void ai_trainer_run(AITrainer* trainer) {

61 printf("--- Starting AI Training Session ---\n");

63 TrainingRule* rule = &trainer->curriculum.rules[i];
64
65 // This simulates the AI processing one rule.
66 printf("Rule %zu: ", i + 1);
67
68 const char* op_symbol;
69 switch (rule->operation) {
70 case OP_KIND_ADD: op_symbol = "+"; break;
71 case OP_KIND_SUBTRACT: op_symbol = "-"; break;
72 case OP_KIND_MULTIPLY: op_symbol = "*"; break;
73 case OP_KIND_DIVIDE: op_symbol = "/"; break;
74 }
75
76 printf("Processing (%lld %s %lld) -> ",
77 rule->input_a.as.int_val,
78 op_symbol,
79 rule->input_b.as.int_val);
80
81 if (rule->result.kind == VAL_KIND_INT) {
82 printf("Result: %lld. ", rule->result.as.int_val);
83 } else {
84 printf("Result: Undefined. ");
85 }
86
87 switch (rule->conflict) {
88 case CONFLICT_TYPE_STABLE:
89 printf("Type: Stable (Liked Pair). Reinforces module %s.\n", rule->resolved_module);
90 break;
91 case CONFLICT_TYPE_UNLIKED_PAIR:
92 printf("Type: Conflict (Unliked Pair). Resolution creates module %s.\n", rule->resolved_module);
93 break;
94 case CONFLICT_TYPE_UNDEFINED:

96 break;
97 }
98 }
