15static int ham_alloc_internal(RegionId* out_id, HamTier tier, size_t size_bytes, void** out_ptr) {
16 if (region_count >= MAX_REGIONS) return -1;
17 void* ptr = nullptr;
18 if (tier == HAM_CRITICAL || tier == HAM_ACTIVE) {
19 ptr = malloc(size_bytes);
20 if (!ptr) return -1;
21 }
22
23 RegionId id = region_count + 1;
24 ham_regions[region_count] = (HamRegion){
25 .id = id,
26 .tier = tier,
27 .capacity_bytes = size_bytes,
28 .base = ptr // nullptr for ARCHIVE tier initially
29 };
30 snprintf(ham_regions[region_count].path, 256, "archive_region_%llu.bin", (unsigned long long)id);
31
32 *out_id = id;
33 if (out_ptr) *out_ptr = ptr;
34 region_count++;
35 return 0;
36}

58static int ham_persist_internal(RegionId id) {
59 if (id == 0 || id > region_count) return -1;
60 HamRegion* region = nullptr;

75 fclose(f);
76 printf("HAM: Persisted Region %llu to %s\n", (unsigned long long)id, region->path);

106// --- Intelligence function implementations ---
107// A simple entropy calculation based on byte value distribution.
108// A real implementation would be more sophisticated.
109static float calculate_entropy(const void* data, size_t size) {
110 if (!data || size == 0) return 0.0f;
111 unsigned int counts[256] = {0};
112 for (size_t i = 0; i < size; i++) {
113 counts[((const uint8_t*)data)[i]]++;
114 }
115
116 float entropy = 0.0f;
117 for (int i = 0; i < 256; i++) {
118 if (counts[i] > 0) {
119 float p = (float)counts[i] / size;
120 entropy -= p * log2f(p);
121 }
122 }
123 return entropy / 8.0f; // Normalize to a 0-1 range (max entropy is 8 bits/byte)
124}

157static int ham_intern_check_internal(RegionId id, MotifDictionary* dict) {
158 for (size_t i = 0; i < region_count; ++i) {
159 if (ham_regions[i].id == id) {
160 HamRegion* region = &ham_regions[i];
161 // Policy: only intern DORMANT regions.
162 if (region->tier == HAM_DORMANT && region->base && !region->interned_motif) {
163 printf("HAM: Attempting to intern Region %llu.\n", (unsigned long long)id);
164 Motif* motif = motif_dict_intern(dict, region->base, region->capacity_bytes);
165 if (motif) {
166 // If the motif was new, ref_count is 1. If it already existed,
167 // ref_count is > 1, meaning we found a duplicate.
168 if (motif->ref_count > 1) {
169 printf("HAM: Found duplicate! Compressing Region %llu.\n", (unsigned long long)id);
170 free(region->base); // Free the original, redundant memory.
171 region->base = motif->data; // Point to the single interned copy.
172 region->interned_motif = motif;
173 } else {
174 // This was the first time seeing this data.
175 region->interned_motif = motif;
176 // We could free the original and point to the motif's copy,
177 // but they are identical, so we just link them.
178 free(motif->data); // The dict made a copy, free it.
179 motif->data = region->base; // Point dict's copy to our memory.
180 }
181 }
182 }
183 return 0;
184 }
185 }
186 return -1;
187}
