212// Architecture detection
213BTL_Architecture btl_detect_architecture(void) {
214#if defined(__x86_64__) || defined(_M_X64) || defined(__amd64__)
215 return BTL_ARCH_X86_64;
216#elif defined(__aarch64__) || defined(_M_ARM64)
217 return BTL_ARCH_ARM64;
218#elif defined(__riscv)
219 return BTL_ARCH_RISCV;
220#else
221 return BTL_ARCH_UNKNOWN;
222#endif
223}
