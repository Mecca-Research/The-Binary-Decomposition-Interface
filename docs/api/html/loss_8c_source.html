92double loss_cross_entropy_forward(const double* predictions, const double* targets,

143// Binary Cross Entropy Loss
144double loss_binary_cross_entropy_forward(const double* predictions, const double* targets,
145 size_t size) {
146 double sum = 0.0;
147 const double epsilon = 1e-7;
148
149 for (size_t i = 0; i < size; i++) {
150 double p = fmax(fmin(predictions[i], 1.0 - epsilon), epsilon);
151 sum -= targets[i] * log(p) + (1.0 - targets[i]) * log(1.0 - p);
152 }
153
154 return sum / size;
155}

205// Huber Loss
206double loss_huber_forward(const double* predictions, const double* targets,
207 size_t size, double delta) {
208 double sum = 0.0;
209 for (size_t i = 0; i < size; i++) {
210 double diff = fabs(predictions[i] - targets[i]);
211 if (diff <= delta) {
212 sum += 0.5 * diff * diff;
213 } else {
214 sum += delta * (diff - 0.5 * delta);
