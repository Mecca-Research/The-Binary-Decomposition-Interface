17// Reports an error at a specific token's location.
18static void error_at(Parser* parser, Token* token, const char* message) {
19 if (parser->panic_mode) return;
20 parser->panic_mode = true;
21 fprintf(stderr, "[line %d] Error", token->line);
22
23 if (token->kind == TOKEN_EOF) {
24 fprintf(stderr, " at end");
25 } else if (token->kind == TOKEN_ERROR) {
26 // Nothing.
27 } else {
28 fprintf(stderr, " at '%.*s'", token->length, token->start);
29 }
30
31 fprintf(stderr, ": %s\n", message);
32 parser->had_error = true;
33}

45// Consumes the current token and advances to the next one.
46static void advance(Parser* parser) {
47 parser->previous = parser->current;
48 for (;;) {
49 parser->current = lexer_scan_token(parser->lexer);
50 if (parser->current.kind != TOKEN_ERROR) break;
51 error_at_current(parser, parser->current.start);
52 }
53}

55// Consumes the current token if it matches the expected kind.
56// Reports an error otherwise.
57static void consume(Parser* parser, TokenKind kind, const char* message) {
58 if (parser->current.kind == kind) {
59 advance(parser);
60 return;
61 }
62 error_at_current(parser, message);
63}

65// Checks if the current token is of the given kind.
66static bool check(Parser* parser, TokenKind kind) {
67 return parser->current.kind == kind;
68}
69
70// If the current token matches the given kind, consumes it and returns true.
71static bool match(Parser* parser, TokenKind kind) {
72 if (!check(parser, kind)) return false;
73 advance(parser);
74 return true;
75}

83// primary = NUMBER | STRING | "true" | "false" | "nil" | "(" expression ")" | IDENTIFIER ;
84static AstNode* primary(Parser* parser) {
85 if (match(parser, TOKEN_FALSE)) return ast_new_literal_int(0); // Representing bools as ints
86 if (match(parser, TOKEN_TRUE)) return ast_new_literal_int(1);
87 if (match(parser, TOKEN_NIL)) return ast_new_literal_int(0); // Or a specific nil type
88
89 if (match(parser, TOKEN_INT_LITERAL)) {
90 long long value = strtoll(parser->previous.start, nullptr, 10);
91 return ast_new_literal_int(value);
92 }
93
94 if (match(parser, TOKEN_FLOAT_LITERAL)) {
95 double value = strtod(parser->previous.start, nullptr);
96 return ast_new_literal_float(value);
97 }
98
99 if (match(parser, TOKEN_LPAREN)) {
100 AstNode* expr = expression(parser);
101 consume(parser, TOKEN_RPAREN, "Expect ')' after expression.");
102 return expr;
103 }
104
105 error(parser, "Expect expression.");
106 return nullptr;
107}

110// unary = ( "!" | "-" ) unary | primary ;
111static AstNode* unary(Parser* parser) {
112 if (match(parser, TOKEN_BANG) || match(parser, TOKEN_MINUS)) {
113 const char* op = parser->previous.kind == TOKEN_BANG ? "!" : "-";
114 AstNode* right = unary(parser);
115 // A proper AST would have ast_new_unary_op. We'll reuse binary for now.
116 return ast_new_binary_op(op, nullptr, right);
117 }
118 return primary(parser);
119}

121// factor = unary ( ( "/" | "*" ) unary )* ;
122static AstNode* factor(Parser* parser) {
123 AstNode* expr = unary(parser);
124 while (match(parser, TOKEN_SLASH) || match(parser, TOKEN_STAR)) {
125 const char* op = parser->previous.kind == TOKEN_SLASH ? "/" : "*";
126 AstNode* right = unary(parser);
127 expr = ast_new_binary_op(op, expr, right);
128 }
129 return expr;
130}

132// term = factor ( ( "-" | "+" ) factor )* ;
133static AstNode* term(Parser* parser) {
134 AstNode* expr = factor(parser);
135 while (match(parser, TOKEN_MINUS) || match(parser, TOKEN_PLUS)) {
136 const char* op = parser->previous.kind == TOKEN_MINUS ? "-" : "+";
137 AstNode* right = factor(parser);
138 expr = ast_new_binary_op(op, expr, right);
139 }
140 return expr;
141}

143// comparison = term ( ( ">" | ">=" | "<" | "<=" ) term )* ;
144static AstNode* comparison(Parser* parser) {
145 AstNode* expr = term(parser);
146 while (match(parser, TOKEN_GREATER) || match(parser, TOKEN_GREATER_EQUAL) ||
147 match(parser, TOKEN_LESS) || match(parser, TOKEN_LESS_EQUAL)) {
148 const char* op;
149 switch (parser->previous.kind) {
150 case TOKEN_GREATER: op = ">"; break;
151 case TOKEN_GREATER_EQUAL: op = ">="; break;
152 case TOKEN_LESS: op = "<"; break;
153 case TOKEN_LESS_EQUAL: op = "<="; break;
154 default: op = ""; // Unreachable
155 }
156 AstNode* right = term(parser);
157 expr = ast_new_binary_op(op, expr, right);
158 }
159 return expr;
160}

162// equality = comparison ( ( "!=" | "==" ) comparison )* ;
163static AstNode* equality(Parser* parser) {
164 AstNode* expr = comparison(parser);
165 while (match(parser, TOKEN_BANG_EQUAL) || match(parser, TOKEN_EQUAL_EQUAL)) {
166 const char* op = parser->previous.kind == TOKEN_BANG_EQUAL ? "!=" : "==";
167 AstNode* right = comparison(parser);
168 expr = ast_new_binary_op(op, expr, right);
169 }
170 return expr;
171}

194// Tries to recover from a syntax error by discarding tokens until a likely
195// statement boundary is found.
196static void synchronize(Parser* parser) {
197 parser->panic_mode = false;
198 while (parser->current.kind != TOKEN_EOF) {
199 if (parser->previous.kind == TOKEN_SEMICOLON) return;
200 switch (parser->current.kind) {
201 case TOKEN_CLASS:
202 case TOKEN_FUN:
203 [[fallthrough]];
204 case TOKEN_VAR:
205 case TOKEN_FOR:
206 case TOKEN_IF:
207 [[fallthrough]];
208 case TOKEN_WHILE:
209 case TOKEN_PRINT:
210 case TOKEN_RETURN:
211 return;
212 default:
213 ; // Do nothing.
214 }
215 advance(parser);
216 }
217}

234AstNode* parser_parse(Parser* parser) {
235 AstNode* program = ast_new_program();
236 if (!program) return nullptr;
237
238 while (!match(parser, TOKEN_EOF)) {
239 AstNode* decl = declaration(parser);
240 if (decl) {
241 bci_vec_push(program->as.block, decl);
242 }
243 if (parser->panic_mode) synchronize(parser);
244 }
245
246 return parser->had_error ? nullptr : program;
247}
