36int wavefront_add_node(Wavefront* wf, NodeId node_id) {
37 if (!wf) return -1;
38
39 if (wf->ready_count >= wf->ready_capacity) {
40 wf->ready_capacity *= 2;
41 wf->ready_nodes = realloc(wf->ready_nodes, sizeof(NodeId) * wf->ready_capacity);
42 if (!wf->ready_nodes) return -1;
43 }
44
45 wf->ready_nodes[wf->ready_count++] = node_id;
46 return 0;

98int scheduler_get_next_wavefront(WavefrontScheduler* sched, Wavefront** out_wavefront) {
99 if (!sched || !out_wavefront) return -1;
100
101 // Build dependency levels
102
103 Wavefront* wf = wavefront_create((uint32_t)sched->wavefront_count);
104 if (!wf) {
105 return -1;
106 }
107
108 // Find nodes with no unvisited dependencies
109 for (size_t i = 0; i < sched->graph->node_count; i++) {
110 if (sched->visited[i]) continue;
111
112 const GraphNode* node = &sched->graph->nodes[i];
113 bool all_deps_ready = true;
114
115 for (size_t j = 0; j < node->input_count; j++) {
116 NodeId input_id = node->inputs[j];
117 if (input_id < sched->graph->node_count && !sched->visited[input_id]) {
118 all_deps_ready = false;
119 break;
120 }
121 }
122
123 if (all_deps_ready) {
124 wavefront_add_node(wf, node->id);
125 sched->visited[i] = true;
126 }
127 }
128
129
130 if (wf->ready_count == 0) {
131 wavefront_free(wf);
132 return -1; // No more wavefronts
133 }
134
135 *out_wavefront = wf;
136 return 0;
137}

139int scheduler_execute_wavefront(WavefrontScheduler* sched, Wavefront* wavefront) {
140 if (!sched || !wavefront) return -1;
141
142 // Execute all nodes in the wavefront in parallel
143 for (size_t i = 0; i < wavefront->ready_count; i++) {
144 NodeId node_id = wavefront->ready_nodes[i];
145 if (node_id >= sched->graph->node_count) continue;
146
147 const GraphNode* node = &sched->graph->nodes[node_id];
148
149 // Select device (simple round-robin for now)
150 DeviceVTable* device = sched->devices[i % sched->device_count];
151
152 // Lower and execute
153 void* kernel = NULL;
154 if (device->lower(node, &kernel) == 0) {
155 // TODO: Prepare HAM regions for execution
156 device->enqueue(kernel, NULL, 0);
157 free(kernel);
158 }
159
160 atomic_fetch_add(&wavefront->completed_count, 1);
161 }
162
163 // Sync all devices
164 for (size_t i = 0; i < sched->device_count; i++) {
165 sched->devices[i]->sync();
166 }
167
168 return 0;
169}

171int wavefront_scheduler_run(WavefrontScheduler* sched) {
172 if (!sched) return -1;
173
174 atomic_store(&sched->running, true);
175
176 while (atomic_load(&sched->running)) {
177 Wavefront* wf = NULL;
178 if (scheduler_get_next_wavefront(sched, &wf) != 0) {
179 break; // No more wavefronts
180 }
181
182 scheduler_execute_wavefront(sched, wf);
183 wavefront_free(wf);
184 }
185
186 atomic_store(&sched->running, false);
187 return 0;
