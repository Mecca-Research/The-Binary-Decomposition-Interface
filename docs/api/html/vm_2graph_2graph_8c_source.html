120Graph* graph_clone(const Graph* graph) {
121 if (!graph) return NULL;
122
123 Graph* clone = graph_create(graph->name);
124 if (!clone) return NULL;
125
126 // Create mapping from original node IDs to cloned node IDs
127 uint32_t* id_mapping = (uint32_t*)malloc(graph->node_count * sizeof(uint32_t));
128 if (!id_mapping) {
129 graph_destroy(clone);
130 return NULL;
131 }
132
133 // Clone nodes and build ID mapping
134 for (uint32_t i = 0; i < graph->node_count; i++) {
135 GraphNode* original = graph->nodes[i];
136 GraphNode* node = graph_add_node(clone, original->type, original->name);
137 if (!node) {
138 free(id_mapping);
139 graph_destroy(clone);
140 return NULL;
141 }
142
143 // Store the mapping from original ID to cloned ID
144 id_mapping[i] = node->id;
145
146 // Copy node data
147 node->constant_value = original->constant_value;
148 if (original->custom_data && original->custom_data_size > 0) {
149 node->custom_data = malloc(original->custom_data_size);
150 if (node->custom_data) {
151 memcpy(node->custom_data, original->custom_data, original->custom_data_size);
152 node->custom_data_size = original->custom_data_size;
153 }
154 }
155 }
156
157 // Clone edges using the ID mapping
158 for (uint32_t i = 0; i < graph->edge_count; i++) {
159 GraphEdge* original = graph->edges[i];
160
161 // Find the indices of source and target nodes in the original graph
162 uint32_t source_idx = UINT32_MAX;
163 uint32_t target_idx = UINT32_MAX;
164
165 for (uint32_t j = 0; j < graph->node_count; j++) {
166 if (graph->nodes[j]->id == original->source->id) {
167 source_idx = j;
168 }
169 if (graph->nodes[j]->id == original->target->id) {
170 target_idx = j;
171 }
172 }
173
174 // Use the mapped IDs to create edges in the cloned graph
175 if (source_idx != UINT32_MAX && target_idx != UINT32_MAX) {
176 uint32_t cloned_source_id = id_mapping[source_idx];
177 uint32_t cloned_target_id = id_mapping[target_idx];
178
179 graph_add_edge(clone, cloned_source_id, original->source_output,
180 cloned_target_id, original->target_input);
181 }
182 }
183
184 free(id_mapping);
185 return clone;
186}

234bool graph_remove_node(Graph* graph, uint32_t node_id) {
235 if (!graph) return false;
236
237 // Find node
238 uint32_t node_index = UINT32_MAX;
239 for (uint32_t i = 0; i < graph->node_count; i++) {
240 if (graph->nodes[i]->id == node_id) {
241 node_index = i;
242 break;
243 }
244 }
245
246 if (node_index == UINT32_MAX) return false;
247
248 GraphNode* node = graph->nodes[node_index];
249
250 // Remove all edges connected to this node
251 for (uint32_t i = 0; i < graph->edge_count; ) {
252 GraphEdge* edge = graph->edges[i];
253 if (edge->source->id == node_id || edge->target->id == node_id) {
254 graph_edge_destroy(edge);
255 // Shift remaining edges
256 for (uint32_t j = i; j < graph->edge_count - 1; j++) {
257 graph->edges[j] = graph->edges[j + 1];
258 }
259 graph->edge_count--;
260 } else {
261 i++;
262 }
263 }
264
265 // Remove from input/output lists
266 if (node->type == GRAPH_NODE_INPUT) {
267 for (uint32_t i = 0; i < graph->input_count; i++) {
268 if (graph->inputs[i] == node) {
269 for (uint32_t j = i; j < graph->input_count - 1; j++) {
270 graph->inputs[j] = graph->inputs[j + 1];
271 }
272 graph->input_count--;
273 break;
274 }
275 }
276 } else if (node->type == GRAPH_NODE_OUTPUT) {
277 for (uint32_t i = 0; i < graph->output_count; i++) {
278 if (graph->outputs[i] == node) {
279 for (uint32_t j = i; j < graph->output_count - 1; j++) {
280 graph->outputs[j] = graph->outputs[j + 1];
281 }
282 graph->output_count--;
283 break;
284 }
285 }
286 }
287
288 // Destroy node
289 graph_node_destroy(node);
290
291 // Shift remaining nodes
292 for (uint32_t i = node_index; i < graph->node_count - 1; i++) {
293 graph->nodes[i] = graph->nodes[i + 1];
294 }
295 graph->node_count--;
296
297 return true;
298}

444bool graph_has_cycles(const Graph* graph) {
445 if (!graph) return false;
446
447 // Use DFS to detect cycles
448 enum { WHITE, GRAY, BLACK };
449 uint8_t* colors = (uint8_t*)calloc(graph->node_count, sizeof(uint8_t));
450 if (!colors) return false;
451
452 bool has_cycle = false;
453
454 // DFS helper function (implemented as iterative to avoid stack overflow)
455 for (uint32_t start = 0; start < graph->node_count && !has_cycle; start++) {
456 if (colors[start] != WHITE) continue;
457
458 // Stack for DFS
459 uint32_t* stack = (uint32_t*)malloc(graph->node_count * sizeof(uint32_t));
460 int stack_top = 0;
461
462 stack[stack_top++] = start;
463 colors[start] = GRAY;
464
465 while (stack_top > 0 && !has_cycle) {
466 uint32_t current_idx = stack[--stack_top];
467 GraphNode* current = graph->nodes[current_idx];
468
469 // Check all outgoing edges
470 for (uint32_t i = 0; i < current->output_count; i++) {
471 GraphEdge* edge = current->outputs[i];
472 if (!edge) continue;
473
474 // Find target node index
475 uint32_t target_idx = UINT32_MAX;
476 for (uint32_t j = 0; j < graph->node_count; j++) {
477 if (graph->nodes[j] == edge->target) {
478 target_idx = j;
479 break;
480 }
481 }
482
483 if (target_idx == UINT32_MAX) continue;
484
485 if (colors[target_idx] == GRAY) {
486 has_cycle = true;
487 break;
488 } else if (colors[target_idx] == WHITE) {
489 colors[target_idx] = GRAY;
490 stack[stack_top++] = target_idx;
491 }
492 }
493
494 colors[current_idx] = BLACK;
495 }
496
497 free(stack);
498 }
499
500 free(colors);
501 return has_cycle;
502}

508uint32_t* graph_topological_sort(const Graph* graph, uint32_t* count) {
509 if (!graph || !count) return NULL;
510
511 *count = 0;
512 if (graph->node_count == 0) return NULL;
513
514 // Kahn's algorithm
515 uint32_t* in_degree = (uint32_t*)calloc(graph->node_count, sizeof(uint32_t));
516 uint32_t* result = (uint32_t*)malloc(graph->node_count * sizeof(uint32_t));
517 uint32_t* queue = (uint32_t*)malloc(graph->node_count * sizeof(uint32_t));
518
519 if (!in_degree || !result || !queue) {
520 free(in_degree);
521 free(result);
522 free(queue);
523 return NULL;
524 }
525
526 // Calculate in-degrees
527 for (uint32_t i = 0; i < graph->node_count; i++) {
528 in_degree[i] = graph->nodes[i]->dependency_count;
529 }
530
531 // Find nodes with no incoming edges
532 uint32_t queue_front = 0, queue_back = 0;
533 for (uint32_t i = 0; i < graph->node_count; i++) {
534 if (in_degree[i] == 0) {
535 queue[queue_back++] = i;
536 }
537 }
538
539 // Process queue
540 while (queue_front < queue_back) {
541 uint32_t current_idx = queue[queue_front++];
542 result[(*count)++] = graph->nodes[current_idx]->id;
543
544 GraphNode* current = graph->nodes[current_idx];
545
546 // Reduce in-degree of neighbors
547 for (uint32_t i = 0; i < current->output_count; i++) {
548 GraphEdge* edge = current->outputs[i];
549 if (!edge) continue;
550
551 // Find target node index
552 uint32_t target_idx = UINT32_MAX;
553 for (uint32_t j = 0; j < graph->node_count; j++) {
554 if (graph->nodes[j] == edge->target) {
555 target_idx = j;
556 break;
557 }
558 }
559
560 if (target_idx != UINT32_MAX) {
561 in_degree[target_idx]--;
562 if (in_degree[target_idx] == 0) {
563 queue[queue_back++] = target_idx;
564 }
565 }
566 }
567 }
568
569 free(in_degree);
570 free(queue);
571
572 // Check if all nodes were processed (no cycles)
573 if (*count != graph->node_count) {
574 free(result);
575 *count = 0;
576 return NULL;
577 }
578
579 return result;
580}

582uint32_t graph_get_depth(const Graph* graph) {
583 if (!graph) return 0;
584
585 uint32_t max_depth = 0;
586
587 // Calculate depth for each node using BFS
588 for (uint32_t i = 0; i < graph->input_count; i++) {
589 GraphNode* input = graph->inputs[i];
590
591 // BFS from this input
592 uint32_t* depths = (uint32_t*)calloc(graph->node_count, sizeof(uint32_t));
593 bool* visited = (bool*)calloc(graph->node_count, sizeof(bool));
594 uint32_t* queue = (uint32_t*)malloc(graph->node_count * sizeof(uint32_t));
595
596 if (!depths || !visited || !queue) {
597 free(depths);
598 free(visited);
599 free(queue);
600 continue;
601 }
602
603 uint32_t queue_front = 0, queue_back = 0;
604
605 // Find input node index
606 uint32_t input_idx = UINT32_MAX;
607 for (uint32_t j = 0; j < graph->node_count; j++) {
608 if (graph->nodes[j] == input) {
609 input_idx = j;
610 break;
611 }
612 }
613
614 if (input_idx == UINT32_MAX) {
615 free(depths);
616 free(visited);
617 free(queue);
618 continue;
619 }
620
621 queue[queue_back++] = input_idx;
622 visited[input_idx] = true;
623 depths[input_idx] = 0;
624
625 while (queue_front < queue_back) {
626 uint32_t current_idx = queue[queue_front++];
627 GraphNode* current = graph->nodes[current_idx];
628
629 for (uint32_t j = 0; j < current->output_count; j++) {
630 GraphEdge* edge = current->outputs[j];
631 if (!edge) continue;
632
633 // Find target node index
634 uint32_t target_idx = UINT32_MAX;
635 for (uint32_t k = 0; k < graph->node_count; k++) {
636 if (graph->nodes[k] == edge->target) {
637 target_idx = k;
638 break;
639 }
640 }
641
642 if (target_idx != UINT32_MAX && !visited[target_idx]) {
643 visited[target_idx] = true;
644 depths[target_idx] = depths[current_idx] + 1;
645 queue[queue_back++] = target_idx;
646
647 if (depths[target_idx] > max_depth) {
648 max_depth = depths[target_idx];
649 }
650 }
651 }
652 }
653
654 free(depths);
655 free(visited);
656 free(queue);
657 }
658
659 return max_depth + 1;
660}

662uint32_t graph_get_width(const Graph* graph) {
663 if (!graph) return 0;
664
665 // Calculate maximum number of nodes at any depth level
666 uint32_t depth = graph_get_depth(graph);
667 if (depth == 0) return 0;
668
669 uint32_t* level_counts = (uint32_t*)calloc(depth, sizeof(uint32_t));
670 if (!level_counts) return 0;
671
672 // Count nodes at each level (simplified approach)
673 for (uint32_t i = 0; i < graph->node_count; i++) {
674 // For simplicity, distribute nodes evenly across levels
675 uint32_t level = i % depth;
676 level_counts[level]++;
677 }
678
679 uint32_t max_width = 0;
680 for (uint32_t i = 0; i < depth; i++) {
681 if (level_counts[i] > max_width) {
682 max_width = level_counts[i];
683 }
684 }
685
686 free(level_counts);
687 return max_width;
688}

690void graph_get_stats(const Graph* graph, GraphStats* stats) {
691 if (!graph || !stats) return;
692
693 memset(stats, 0, sizeof(GraphStats));
694
695 stats->total_nodes = graph->node_count;
696 stats->total_edges = graph->edge_count;
697 stats->input_nodes = graph->input_count;
698 stats->output_nodes = graph->output_count;
699 stats->total_executions = graph->execution_count;
700 stats->total_execution_time_ns = graph->total_execution_time_ns;
701
702 if (stats->total_executions > 0) {
703 stats->average_execution_time_ns =
704 (double)stats->total_execution_time_ns / stats->total_executions;
705 }
706
707 // Count node types
708 for (uint32_t i = 0; i < graph->node_count; i++) {
709 GraphNode* node = graph->nodes[i];
710
711 switch (node->type) {
712 case GRAPH_NODE_CONSTANT:
713 stats->constant_nodes++;
714 break;
715 case GRAPH_NODE_INPUT:
716 case GRAPH_NODE_OUTPUT:
717 // Already counted above
718 break;
719 case GRAPH_NODE_BRANCH:
720 case GRAPH_NODE_PHI:
721 case GRAPH_NODE_MERGE:
722 case GRAPH_NODE_LOOP_BEGIN:
723 case GRAPH_NODE_LOOP_END:
724 stats->control_nodes++;
725 break;
726 default:
727 stats->operation_nodes++;
728 break;
729 }
730
731 if (node->is_dead) {
732 stats->dead_nodes++;
733 }
734 }
735}

890// Memory management helpers
891void* graph_alloc(Graph* graph, size_t size) {
892 if (!graph || !graph->memory_pool) return malloc(size);
893
894 // Align to 8-byte boundary
895 size = (size + 7) & ~7;
896
897 if (graph->memory_used + size > graph->memory_pool_size) {
898 // Fall back to regular malloc if pool is full
899 return malloc(size);
900 }
901
902 void* ptr = (char*)graph->memory_pool + graph->memory_used;
903 graph->memory_used += size;
904 return ptr;
905}

907void graph_free(Graph* graph, void* ptr) {
908 if (!graph || !ptr) return;
909
910 // Check if pointer is in memory pool
911 if (ptr >= graph->memory_pool &&
912 ptr < (char*)graph->memory_pool + graph->memory_pool_size) {
913 // Memory pool allocation - no individual free
914 return;
915 }
916
917 // Regular malloc allocation
918 free(ptr);
919}

957Graph* graph_load_from_file(const char* filename) {
958 if (!filename) return NULL;
959
960 FILE* file = fopen(filename, "r");
961 if (!file) return NULL;
962
963 Graph* graph = graph_create("loaded");
964 if (!graph) {
965 fclose(file);
966 return NULL;
967 }
968
969 char line[256];
970 while (fgets(line, sizeof(line), file)) {
971 if (line[0] == '#') continue; // Comment
972
973 if (strncmp(line, "node", 4) == 0) {
974 uint32_t id;
975 char type_str[32], name[64];
976 if (sscanf(line, "node %u %s %s", &id, type_str, name) == 3) {
977 // Find node type
978 GraphNodeType type = GRAPH_NODE_CUSTOM;
979 for (int t = 0; t <= GRAPH_NODE_CUSTOM; t++) {
980 if (strcmp(graph_node_type_to_string(t), type_str) == 0) {
981 type = t;
982 break;
983 }
984 }
985 graph_add_node(graph, type, name);
986 }
987 } else if (strncmp(line, "edge", 4) == 0) {
988 uint32_t src_id, src_out, tgt_id, tgt_in;
989 if (sscanf(line, "edge %u %u %u %u", &src_id, &src_out, &tgt_id, &tgt_in) == 4) {
990 graph_add_edge(graph, src_id, src_out, tgt_id, tgt_in);
991 }
992 }
993 }
994
995 fclose(file);
996 return graph;
997}
