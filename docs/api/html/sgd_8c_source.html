37void sgd_step(SGDOptimizer* opt, double* params, double* gradients) {
38 if (!opt || !params || !gradients) return;
39
40 opt->step_count++;
41
42 for (size_t i = 0; i < opt->num_params; i++) {
43 double grad = gradients[i];
44
45 // Apply weight decay
46 if (opt->config.weight_decay > 0.0) {
47 grad += opt->config.weight_decay * params[i];
48 }
49
50 // Update velocity with momentum
51 if (opt->config.momentum > 0.0) {
52 opt->velocity[i] = opt->config.momentum * opt->velocity[i] + grad;
53
54 // Nesterov momentum
55 if (opt->config.nesterov) {
56 grad = grad + opt->config.momentum * opt->velocity[i];
57 } else {
58 grad = opt->velocity[i];
59 }
60 }
61
62 // Update parameters
63 params[i] -= opt->config.learning_rate * grad;
64 }
65}
