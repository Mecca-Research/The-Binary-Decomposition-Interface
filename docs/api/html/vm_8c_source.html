19EnhancedVM* enhanced_vm_create(size_t heap_size) {
20 // Split heap: 10% nursery, 90% old generation

22 size_t old_gen_size = heap_size - nursery_size;
23 return enhanced_vm_create_with_sizes(nursery_size, old_gen_size);

26EnhancedVM* enhanced_vm_create_with_sizes(size_t nursery_size, size_t old_gen_size) {
27 EnhancedVM* vm = (EnhancedVM*)malloc(sizeof(EnhancedVM));
28 if (!vm) return NULL;
29
30 // Initialize base VM
31 vm->base_vm = (VM*)malloc(sizeof(VM));
32 if (!vm->base_vm) {
33 free(vm);
34 return NULL;
35 }
36 vm_init(vm->base_vm);
37
38 // Initialize GC
39 vm->gc = generational_gc_create(nursery_size, old_gen_size, DEFAULT_PERM_SIZE);
40 if (!vm->gc) {
41 free(vm->base_vm);
42 free(vm);
43 return NULL;
44 }
45
46 // Initialize root set
47 vm->gc_roots = gc_root_set_create();
48 if (!vm->gc_roots) {
49 generational_gc_destroy(vm->gc);
50 free(vm->base_vm);
51 free(vm);
52 return NULL;
53 }
54
55 // Initialize JIT components (NULL for now)
56 vm->jit_compiler = NULL;
57 vm->hot_path_detector = NULL;
58 vm->tiered_compilation = NULL;
59
60 // Configuration
61 vm->enable_jit = false;
62 vm->enable_gc = true;
63 vm->enable_profiling = false;
64 vm->gc_threshold = 80; // Trigger at 80% full
65
66 // Statistics
67 vm->total_executions = 0;
68 vm->jit_executions = 0;
69 vm->interpreter_executions = 0;
70 vm->gc_collections = 0;
71 vm->gc_bytes_allocated = 0;
72 vm->gc_bytes_freed = 0;
73
74 return vm;
75}

102void* vm_alloc(EnhancedVM* vm, size_t size) {
103 if (!vm || !vm->gc) return NULL;
104
105 // Try to allocate from GC
106 GenObject* obj = generational_gc_allocate(vm->gc, size, 0, GEN_YOUNG);
107
108 if (!obj) {
109 // Allocation failed, trigger GC and retry
110 vm_gc_collect(vm);
111 obj = generational_gc_allocate(vm->gc, size, 0, GEN_YOUNG);
112
113 if (!obj) {
114 // Still failed, out of memory
115 fprintf(stderr, "VM: Out of memory (requested %zu bytes)\n", size);
116 return NULL;
117 }
118 }
119
120 vm->gc_bytes_allocated += size;
121 return obj;
122}

144void vm_gc_collect(EnhancedVM* vm) {
145 if (!vm || !vm->gc || !vm->enable_gc) return;
146
147 // Collect roots from VM
148 vm_collect_roots(vm, vm->gc_roots);
149
150 // Get stats before collection
151 size_t young_before = vm->gc->generations[GEN_YOUNG].used;
152 size_t old_before = vm->gc->generations[GEN_OLD].used;
153
154 // Trigger minor GC
155 generational_gc_minor_collect(vm->gc, vm->gc_roots);
156
157 // Update statistics
158 vm->gc_collections++;
159
160 // Calculate freed bytes
161 size_t young_after = vm->gc->generations[GEN_YOUNG].used;
162 size_t old_after = vm->gc->generations[GEN_OLD].used;
163
164 if (young_before > young_after) {
165 vm->gc_bytes_freed += (young_before - young_after);
166 }
167
168 // Check if major GC is needed
169 GenerationSpace* old_gen = &vm->gc->generations[GEN_OLD];
170 if (old_gen->used > old_gen->size * 0.8) {
171 generational_gc_major_collect(vm->gc, vm->gc_roots);
172 vm->gc_collections++;
173
174 if (old_before > old_after) {
175 vm->gc_bytes_freed += (old_before - old_after);
176 }
177 }
178}

180void vm_write_barrier(EnhancedVM* vm, void* old_obj, void* new_value) {
181 if (!vm || !vm->gc) return;
182
183 // Guard against NULL operands - legitimate cases include:
184 // - Clearing a field: obj->field = NULL
185 // - Holder is absent: NULL->field = obj
186 if (!old_obj || !new_value) return;
187
188 GenObject* old = (GenObject*)old_obj;
189 GenObject* new = (GenObject*)new_value;
190
191 // Check if this is an old→young pointer
192 if (old->header.generation == GEN_OLD &&
193 new->header.generation == GEN_YOUNG) {
194 // Add to remembered set
195 generational_gc_write_barrier(vm->gc, old, new);
196 }
197}
