72void ast_pattern_free(AstPattern* pattern) {
73 if (!pattern) return;
74
75 switch (pattern->kind) {
76 case PATTERN_BINDING:
77 free((void*)pattern->as.binding.name);
78 break;
79 case PATTERN_STRUCT:
80 free((void*)pattern->as.struct_pattern.struct_name);
81 for (size_t i = 0; i < pattern->as.struct_pattern.fields.len; i++) {
82 ast_pattern_free(pattern->as.struct_pattern.fields.data[i]);
83 }
84 bci_vec_free(&pattern->as.struct_pattern.fields);
85 break;
86 case PATTERN_LITERAL:
87 if (pattern->as.literal.value) {
88 ast_free_node(pattern->as.literal.value);
89 }
90 break;
91 default:
92 break;
93 }
94
95 free(pattern);
96}
