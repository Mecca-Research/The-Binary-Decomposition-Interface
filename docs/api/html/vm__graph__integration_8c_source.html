16static uint64_t get_time_ns(void) {
17 struct timespec ts;
18 clock_gettime(CLOCK_MONOTONIC, &ts);
19 return (uint64_t)ts.tv_sec * 1000000000ULL + (uint64_t)ts.tv_nsec;

22// Hash function for cache
23static uint32_t hash_function_id(uint32_t function_id) {
24 // Simple hash function
25 function_id ^= function_id >> 16;
26 function_id *= 0x85ebca6b;
27 function_id ^= function_id >> 13;
28 function_id *= 0xc2b2ae35;
29 function_id ^= function_id >> 16;
30 return function_id % GRAPH_CACHE_BUCKET_COUNT;
31}

52
53 // Free all entries

68Graph* graph_cache_get(GraphCache* cache, uint32_t function_id) {
69 if (!cache) return NULL;
70
71 uint32_t bucket = hash_function_id(function_id);
72 GraphCacheEntry* entry = cache->buckets[bucket];
73
74 while (entry) {
75 if (entry->function_id == function_id) {
76 entry->last_used = get_time_ns();
77 cache->cache_hits++;
78 return entry->graph;
79 }
80 entry = entry->next;
81 }
82
83 cache->cache_misses++;

87bool graph_cache_put(GraphCache* cache, uint32_t function_id, Graph* graph) {
88 if (!cache || !graph) return false;
89
90 // Check if entry already exists
91 if (graph_cache_get(cache, function_id)) {
92 return true; // Already cached
93 }
94
95 // Check if we need to evict
96 if (cache->entry_count >= cache->max_entries) {
97 graph_cache_evict_lru(cache);
98 }
99
100 // Create new entry
101 GraphCacheEntry* entry = (GraphCacheEntry*)calloc(1, sizeof(GraphCacheEntry));
102 if (!entry) return false;
103
104 entry->function_id = function_id;
105 entry->graph = graph;
106 entry->creation_time = get_time_ns();
107 entry->last_used = entry->creation_time;
108
109 // Add to bucket
110 uint32_t bucket = hash_function_id(function_id);
111 entry->next = cache->buckets[bucket];
112 cache->buckets[bucket] = entry;
113 cache->entry_count++;
114
115 return true;
116}

118void graph_cache_remove(GraphCache* cache, uint32_t function_id) {
119 if (!cache) return;
120
121 uint32_t bucket = hash_function_id(function_id);
122 GraphCacheEntry** entry_ptr = &cache->buckets[bucket];
123
124 while (*entry_ptr) {
125 GraphCacheEntry* entry = *entry_ptr;
126 if (entry->function_id == function_id) {
127 *entry_ptr = entry->next;
128 graph_destroy(entry->graph);
129 free(entry);
130 cache->entry_count--;

154void graph_cache_evict_lru(GraphCache* cache) {
155 if (!cache || cache->entry_count == 0) return;
156
157 // Find LRU entry
158 GraphCacheEntry* lru_entry = NULL;
159 GraphCacheEntry** lru_entry_ptr = NULL;
160 uint64_t oldest_time = UINT64_MAX;
161
162 for (size_t i = 0; i < cache->bucket_count; i++) {
163 GraphCacheEntry** entry_ptr = &cache->buckets[i];
164 while (*entry_ptr) {
165 GraphCacheEntry* entry = *entry_ptr;
166 if (entry->last_used < oldest_time) {
167 oldest_time = entry->last_used;
168 lru_entry = entry;
169 lru_entry_ptr = entry_ptr;
170 }
171 entry_ptr = &entry->next;
172 }
173 }
174
175 // Remove LRU entry
176 if (lru_entry && lru_entry_ptr) {
177 *lru_entry_ptr = lru_entry->next;
178 graph_destroy(lru_entry->graph);
179 free(lru_entry);
180 cache->entry_count--;
181 cache->evictions++;
182 }

243GraphVMResult graph_vm_execute(GraphIntegratedVM* vm, const Chunk* chunk) {
244 GraphVMResult result = {0};
245
246 if (!vm || !chunk) {
247 result.error_message = strdup("Invalid VM or chunk");
248 return result;
249 }
250
251 uint64_t start_time = get_time_ns();
252
253 // Decide execution strategy based on configuration
254 switch (vm->config.mode) {
255 case GRAPH_VM_MODE_GRAPH_ONLY: {
256 // Convert to graph and execute
257 Graph* graph = graph_vm_convert_bytecode_to_graph(vm, chunk);
258 if (graph) {
259 result = graph_vm_execute_graph(vm, graph, NULL, 0);
260 result.used_graph_execution = true;
261 graph_destroy(graph);
262 } else {
263 result.error_message = strdup("Failed to convert bytecode to graph");
264 }
265 break;
266 }
267
268 case GRAPH_VM_MODE_BYTECODE_ONLY: {
269 // Execute using base JIT VM
270 JITVmResult jit_result = jit_vm_execute(vm->base_vm, chunk);
271 result.success = jit_result.success;
272 result.used_jit_compilation = jit_result.used_jit;
273 result.execution_time_ns = jit_result.execution_time_ns;
274 result.compilation_time_ns = jit_result.compilation_time_ns;
275
276 if (result.success) {
277 result.output_count = 1;
278 result.output_values = (GraphValue*)malloc(sizeof(GraphValue));
279 if (result.output_values) {
280 result.output_values[0].type = GRAPH_TYPE_F64;
281 result.output_values[0].data.f64 = jit_result.result_value;
282 }
283 } else {
284 result.error_message = strdup("Bytecode execution failed");
285 }
286 break;
287 }
288
289 case GRAPH_VM_MODE_HYBRID:
290 case GRAPH_VM_MODE_ADAPTIVE:
291 default: {
292 // Try graph execution first, fall back to bytecode
293 Graph* graph = graph_vm_convert_bytecode_to_graph(vm, chunk);
294 if (graph) {
295 result = graph_vm_execute_graph(vm, graph, NULL, 0);
296 result.used_graph_execution = true;
297
298 if (!result.success) {
299 // Fall back to bytecode execution
300 JITVmResult jit_result = jit_vm_execute(vm->base_vm, chunk);
301 result.success = jit_result.success;
302 result.used_jit_compilation = jit_result.used_jit;
303 result.used_graph_execution = false;
304
305 if (result.success && !result.output_values) {
306 result.output_count = 1;
307 result.output_values = (GraphValue*)malloc(sizeof(GraphValue));
308 if (result.output_values) {
309 result.output_values[0].type = GRAPH_TYPE_F64;
310 result.output_values[0].data.f64 = jit_result.result_value;
311 }
312 }
313 }
314
315 graph_destroy(graph);
316 } else {
317 // Execute using base JIT VM
318 JITVmResult jit_result = jit_vm_execute(vm->base_vm, chunk);
319 result.success = jit_result.success;
320 result.used_jit_compilation = jit_result.used_jit;
321 result.execution_time_ns = jit_result.execution_time_ns;
322 result.compilation_time_ns = jit_result.compilation_time_ns;
323
324 if (result.success) {
325 result.output_count = 1;
326 result.output_values = (GraphValue*)malloc(sizeof(GraphValue));
327 if (result.output_values) {
328 result.output_values[0].type = GRAPH_TYPE_F64;
329 result.output_values[0].data.f64 = jit_result.result_value;
330 }
331 }
332 }
333 break;
334 }
335 }
336
337 // Update statistics
338 vm->total_functions_executed++;
339 if (result.used_graph_execution) {
340 vm->graph_executions++;
341 vm->graph_execution_time_ns += result.execution_time_ns;
342 } else {
343 vm->bytecode_executions++;
344 vm->bytecode_execution_time_ns += result.execution_time_ns;
345 }
346
347 if (!result.execution_time_ns) {
348 result.execution_time_ns = get_time_ns() - start_time;
349 }
350
351 return result;
352}

406GraphVMResult graph_vm_execute_graph(GraphIntegratedVM* vm, Graph* graph,
407 GraphValue* inputs, uint32_t input_count) {
408 GraphVMResult result = {0};
409
410 if (!vm || !graph) {
411 result.error_message = strdup("Invalid VM or graph");
412 return result;
413 }
414
415 uint64_t start_time = get_time_ns();
416
417 // Optimize graph if enabled
418 if (vm->config.enable_graph_optimization && !graph->is_optimized) {
419 uint64_t opt_start = get_time_ns();
420 graph_optimizer_optimize(vm->graph_optimizer, graph);
421 vm->graph_optimization_time_ns += get_time_ns() - opt_start;
422 vm->graph_optimizations++;
423 }
424
425 // Execute graph
426 GraphExecutionResult exec_result = graph_executor_execute(vm->graph_executor, graph,
427 inputs, input_count);
428
429 result.success = exec_result.success;
430 result.output_values = exec_result.output_values;
431 result.output_count = exec_result.output_count;
432 result.execution_time_ns = exec_result.execution_time_ns;
433 result.nodes_executed = exec_result.nodes_executed;
434 result.used_graph_execution = true;
435
436 if (!result.success && exec_result.error_message) {
437 result.error_message = strdup(exec_result.error_message);
438 }
439
440 // Don't free exec_result.output_values as we're transferring ownership
441 free(exec_result.error_message);
442
443 if (!result.execution_time_ns) {
444 result.execution_time_ns = get_time_ns() - start_time;
445 }
446
447 return result;
448}

450// Bytecode to graph conversion (simplified implementation)
451Graph* graph_vm_convert_bytecode_to_graph(GraphIntegratedVM* vm, const Chunk* chunk) {
452 if (!vm || !chunk) return NULL;
453
454 // Create graph builder
455 GraphBuilder* builder = graph_builder_create("bytecode_graph");
456 if (!builder) return NULL;
457
458 // Simple conversion: create a linear chain of operations
459 // This is a very basic implementation - a full implementation would
460 // parse the bytecode and create appropriate graph nodes
461
462 uint32_t input_node = graph_builder_add_input(builder, GRAPH_TYPE_F64, "input");
463 uint32_t current_node = input_node;
464
465 // Process bytecode instructions (simplified)
466 for (int i = 0; i < chunk->count; i++) {
467 uint8_t instruction = chunk->code[i];
468
469 switch (instruction) {
470 case OP_CONSTANT: {
471 if (i + 1 < chunk->count) {
472 uint8_t constant_index = chunk->code[++i];
473 if (constant_index < chunk->constants.count) {
474 double value = chunk->constants.values[constant_index];
475 GraphValue const_val = {0};
476 const_val.type = GRAPH_TYPE_F64;
477 const_val.data.f64 = value;
478 current_node = graph_builder_add_constant(builder, const_val, "constant");
479 }
480 }
481 break;
482 }
483
484 case OP_ADD: {
485 // For simplicity, add with previous node
486 uint32_t add_node = graph_builder_add(builder, current_node, current_node, "add");
487 if (add_node) current_node = add_node;
488 break;
489 }
490
491 case OP_SUBTRACT: {
492 uint32_t sub_node = graph_builder_sub(builder, current_node, current_node, "sub");
493 if (sub_node) current_node = sub_node;
494 break;
495 }
496
497 case OP_MULTIPLY: {
498 uint32_t mul_node = graph_builder_mul(builder, current_node, current_node, "mul");
499 if (mul_node) current_node = mul_node;
500 break;
501 }
502
503 case OP_DIVIDE: {
504 uint32_t div_node = graph_builder_div(builder, current_node, current_node, "div");
505 if (div_node) current_node = div_node;
506 break;
507 }
508
509 default:
510 // Skip unknown instructions
511 break;
512 }
513 }
514
515 // Add output node
516 uint32_t output_node = graph_builder_add_output(builder, GRAPH_TYPE_F64, "output");
517 if (output_node && current_node) {
518 graph_builder_connect(builder, current_node, 0, output_node, 0);
519 }
520
521 // Build the graph
522 Graph* graph = graph_builder_build(builder);
523 graph_builder_destroy(builder);
524
525 return graph;
526}
