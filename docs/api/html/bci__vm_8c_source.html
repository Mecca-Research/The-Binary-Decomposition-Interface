35static BciVmResult run_with_result(VM* vm) {
36 BciVmResult result;
37 result.status = INTERPRET_OK;
38 result.result_value = 0.0;
39
40#define READ_BYTE() (*vm->ip++)
41#define READ_CONSTANT() (vm->chunk->constants.data[READ_BYTE()])
42
43// Macro to handle binary operations to reduce code duplication.
44#define BINARY_OP(op) \
45 do { \
46 double b = vm_stack_pop(vm); \
47 double a = vm_stack_pop(vm); \
48 vm_stack_push(vm, a op b); \
49 } while (false)
50
51
52 for (;;) {
53 uint8_t instruction;
54 switch (instruction = READ_BYTE()) {
55 case OP_RETURN: {
56 // Capture the result value BEFORE popping it
57 if (vm->stack_top > vm->stack) {
58 result.result_value = vm->stack_top[-1]; // Peek at top value
59 }
60 double value = vm_stack_pop(vm);
61 printf("Result: %g\n", value);
62 result.status = INTERPRET_OK;
63 return result;
64 }
65 case OP_CONSTANT: {
66 double constant = READ_CONSTANT();
67 vm_stack_push(vm, constant);
68 break;
69 }
70 case OP_NEGATE: {
71 vm_stack_push(vm, -vm_stack_pop(vm));
72 break;
73 }
74 case OP_ADD: BINARY_OP(+); break;
75 case OP_SUBTRACT: BINARY_OP(-); break;
76 case OP_MULTIPLY: BINARY_OP(*); break;
77 case OP_DIVIDE: BINARY_OP(/); break;
78
79 default:
80 // Handle unknown opcode
81 result.status = INTERPRET_RUNTIME_ERROR;
82 return result;
83 }
84 }
85
86#undef READ_BYTE
87#undef READ_CONSTANT
88#undef BINARY_OP
89}
